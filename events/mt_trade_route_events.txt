# 0002: Caravan delayed by weather
mt_route.0002 = {
	type = character_event
	title = mt_route.0002.t
	desc = mt_route.0002.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = worry
	}

	trigger = {
		OR = {
			has_trait = wandering_merchant
			has_trait = trade_post_owner
			has_trait = trading_company_head
		}
		NOT = { has_character_modifier = mt_weather_delayed }
	}

	mean_time_to_happen = {
		months = 10

		# More common in winter
		modifier = {
			factor = 0.5
			current_month >= 11
		}

		modifier = {
			factor = 0.5
			current_month <= 2
		}
	}

	immediate = {
		# Apply temporary delay
		add_character_modifier = {
			modifier = mt_weather_delayed
			months = 3
		}
	}

	# Wait it out patiently
	option = {
		name = mt_route.0002.a

		add_stress = 10

		# Chance to gain patient
		random = {
			chance = 15
			add_trait = patient
		}

		ai_chance = {
			base = 60

			modifier = {
				add = 30
				has_trait = patient
			}
		}
	}

	# Try to push through anyway
	option = {
		name = mt_route.0002.b

		trigger = {
			gold >= 50
		}

		remove_short_term_gold = 50

		random_list = {
			60 = { # Make it through
				desc = mt_route.0002.b.success
				remove_character_modifier = mt_weather_delayed
				add_gold = 100
				add_prestige = 35
			}
			40 = { # Lose goods trying
				desc = mt_route.0002.b.failure
				add_gold = -75
				add_stress = 25
			}
		}

		ai_chance = {
			base = 30

			modifier = {
				add = 40
				has_trait = brave
			}

			modifier = {
				add = 20
				gold >= 200
			}
		}
	}

	# Hire local guides
	option = {
		name = mt_route.0002.c

		trigger = {
			gold >= 75
			learning >= 10
		}

		remove_short_term_gold = 75
		remove_character_modifier = mt_weather_delayed

		# Successfully navigate with expert help
		add_gold = 80
		add_prestige = 25
		add_learning = 1

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				learning >= 12
			}

			modifier = {
				add = 20
				gold >= 300
			}
		}
	}
}

# 0003: Bandits attack caravan
mt_route.0003 = {
	type = character_event
	title = mt_route.0003.t
	desc = mt_route.0003.desc
	theme = martial

	left_portrait = {
		character = root
		animation = worry
	}

	right_portrait = {
		character = scope:bandit_leader
		animation = anger
	}

	trigger = {
		OR = {
			has_trait = wandering_merchant
			has_trait = trade_post_owner
			has_trait = trading_company_head
		}
		NOT = { has_character_modifier = mt_protected_operations }
		NOT = { has_character_flag = recent_bandit_attack }
	}

	mean_time_to_happen = {
		months = 12

		# Less likely if you have good reputation
		modifier = {
			factor = 1.5
			prestige >= 1000
		}

		# More likely if at war nearby
		modifier = {
			factor = 0.6
			any_neighboring_realm = {
				is_at_war = yes
			}
		}
	}

	immediate = {
		add_character_flag = {
			flag = recent_bandit_attack
			days = 365
		}

		# Create bandit leader
		create_character = {
			location = root.location
			template = bandit_character
			save_scope_as = bandit_leader
		}
	}

	# Pay the ransom
	option = {
		name = mt_route.0003.a

		trigger = {
			gold >= 100
		}

		remove_short_term_gold = 100
		add_prestige = -25
		add_stress = 15

		# Route continues safely
		custom_tooltip = mt_route.0003.a.tt

		scope:bandit_leader = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
		}

		ai_chance = {
			base = 50

			modifier = {
				add = 30
				has_trait = craven
			}

			modifier = {
				add = 20
				gold >= 500
			}
		}
	}

	# Fight back!
	option = {
		name = mt_route.0003.b

		trigger = {
			martial >= 8
		}

		duel = {
			skill = martial
			target = scope:bandit_leader

			60 = { # Victory
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
				}
				desc = mt_route.0003.b.success

				send_interface_toast = {
					title = mt_route.0003.b.success
					left_icon = root

					add_prestige = 150
					add_gold = 75

					# Gain reputation for being tough
					add_character_modifier = {
						modifier = mt_defended_caravan
						years = 5
					}

					# Bandit leader flees or dies
					scope:bandit_leader = {
						death = {
							death_reason = death_battle
							killer = root
						}
					}
				}
			}

			40 = { # Defeat
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3
				}
				desc = mt_route.0003.b.failure

				send_interface_toast = {
					title = mt_route.0003.b.failure
					left_icon = root

					add_gold = -200
					add_prestige = -75
					add_stress = 35

					# Possible injury
					random = {
						chance = 30
						add_trait = wounded
					}
				}
			}
		}

		ai_chance = {
			base = 30

			modifier = {
				add = 50
				martial >= 12
			}

			modifier = {
				add = 40
				has_trait = brave
			}

			modifier = {
				add = -50
				has_trait = craven
			}
		}
	}

	# Hire guards for future protection
	option = {
		name = mt_route.0003.c

		trigger = {
			gold >= 150
		}

		remove_short_term_gold = 150

		# Lose goods this time
		add_gold = -100
		add_stress = 20

		# But protected in future
		add_character_modifier = {
			modifier = mt_protected_operations
			years = 10
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				gold >= 500
			}

			modifier = {
				add = 20
				stewardship >= 12
			}
		}
	}

	# Try to negotiate
	option = {
		name = mt_route.0003.d

		trigger = {
			diplomacy >= 10
		}

		duel = {
			skill = diplomacy
			target = scope:bandit_leader

			50 = { # Successful negotiation
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
				}
				desc = mt_route.0003.d.success

				send_interface_toast = {
					title = mt_route.0003.d.success
					left_icon = root

					remove_short_term_gold = 50
					add_prestige = 50

					# They let you pass with minimal loss
					scope:bandit_leader = {
						add_opinion = {
							target = root
							modifier = respectful_opinion
							opinion = 30
						}
					}
				}
			}

			50 = { # Failed negotiation
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3
				}
				desc = mt_route.0003.d.failure

				send_interface_toast = {
					title = mt_route.0003.d.failure
					left_icon = root

					add_gold = -150
					add_prestige = -50
					add_stress = 30
				}
			}
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 40
				diplomacy >= 14
			}
		}
	}
}

# 0005: Local lord demands toll

		character = root

		character = scope:toll_lord

			is_landed = yes
