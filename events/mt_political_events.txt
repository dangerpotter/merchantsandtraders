# 0001: Ruler demands "loan" (shakedown)
mt_politics.0001 = {
	type = character_event
	title = mt_politics.0001.t
	desc = mt_politics.0001.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = worry
	}

	right_portrait = {
		character = scope:demanding_ruler
		animation = personality_bold
	}

	trigger = {
		OR = {
			has_trait = wandering_merchant
			has_trait = trade_post_owner
			has_trait = trading_company_head
			has_trait = merchant_prince
		}
		gold >= 100
		exists = liege
		liege = {
			gold < 200
			NOT = { has_relation_friend = root }
		}
	}

	mean_time_to_happen = {
		months = 48

		modifier = {
			factor = 0.6
			gold >= 500
		}

		modifier = {
			factor = 0.7
			liege = { has_trait = greedy }
		}

		modifier = {
			factor = 1.5
			liege = { has_trait = just }
		}
	}

	immediate = {
		liege = { save_scope_as = demanding_ruler }
	}

	# Pay the "loan"
	option = {
		name = mt_politics.0001.a

		remove_short_term_gold = 150

		scope:demanding_ruler = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
		}

		add_prestige = -50
		add_stress = 15

		ai_chance = {
			base = 50

			modifier = {
				add = 30
				has_trait = craven
			}

			modifier = {
				add = 20
				gold >= 300
			}
		}
	}

	# Refuse and risk consequences
	option = {
		name = mt_politics.0001.b

		trigger = {
			gold >= 200
		}

		scope:demanding_ruler = {
			add_opinion = {
				target = root
				modifier = refused_request_opinion
				opinion = -40
			}
		}

		add_prestige = 100
		add_stress = 30

		# Risk of arrest or confiscation
		random = {
			chance = 40
			trigger_event = {
				id = mt_politics.0024
				days = { 30 90 }
			}
		}

		ai_chance = {
			base = 25

			modifier = {
				add = 40
				has_trait = brave
			}

			modifier = {
				add = 30
				has_trait = arrogant
			}

			modifier = {
				add = -20
				gold < 500
			}
		}
	}

	# Negotiate a smaller amount
	option = {
		name = mt_politics.0001.c

		trigger = {
			diplomacy >= 12
		}

		remove_short_term_gold = 75

		scope:demanding_ruler = {
			add_opinion = {
				target = root
				modifier = pleased_opinion
				opinion = 10
			}
		}

		add_diplomacy_lifestyle_xp = 50
		add_stress = -10

		ai_chance = {
			base = 60

			modifier = {
				add = 30
				diplomacy >= 14
			}

			modifier = {
				add = 20
				has_trait = diplomat
			}
		}
	}
}

# 0003: Ruler seizing merchant assets
mt_politics.0003 = {
	type = character_event
	title = mt_politics.0003.t
	desc = mt_politics.0003.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = shock
	}

	right_portrait = {
		character = scope:seizing_ruler
		animation = personality_bold
	}

	trigger = {
		OR = {
			has_trait = wandering_merchant
			has_trait = trade_post_owner
			has_trait = trading_company_head
			has_trait = merchant_prince
		}
		gold >= 150
		exists = liege
		liege = {
			gold < 100
			OR = {
				has_trait = greedy
				has_trait = ambitious
				is_at_war = yes
			}
		}
	}

	mean_time_to_happen = {
		months = 72

		modifier = {
			factor = 0.5
			liege = { has_trait = arbitrary }
		}

		modifier = {
			factor = 2.0
			liege = { has_trait = just }
		}
	}

	immediate = {
		liege = { save_scope_as = seizing_ruler }
	}

	# Accept the seizure quietly
	option = {
		name = mt_politics.0003.a

		remove_short_term_gold = 200
		add_prestige = -100
		add_stress = 40

		# At least you stay safe
		scope:seizing_ruler = {
			add_opinion = {
				target = root
				modifier = compliant_opinion
				opinion = 10
			}
		}

		ai_chance = {
			base = 30

			modifier = {
				add = 40
				has_trait = craven
			}
		}
	}

	# Organize resistance with other merchants
	option = {
		name = mt_politics.0003.b

		trigger = {
			intrigue >= 12
			gold >= 300
		}

		remove_short_term_gold = 150

		scope:seizing_ruler = {
			add_opinion = {
				target = root
				modifier = resisted_opinion
				opinion = -30
			}
		}

		add_prestige = 150

		# Resistance may succeed
		random = {
			chance = 40
			add_gold = 200
			trigger_event = {
				id = mt_politics.0018
				days = 90
			}
		}

		random = {
			chance = 60
			add_stress = 30
			remove_short_term_gold = 100
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				has_trait = brave
			}

			modifier = {
				add = 20
				intrigue >= 14
			}
		}
	}

	# Flee to another realm
	option = {
		name = mt_politics.0003.c

		trigger = {
			any_neighboring_realm_same_rank_owner = {
				NOT = { this = scope:seizing_ruler }
			}
		}

		remove_short_term_gold = 100
		add_prestige = -50

		# Move to new realm
		random_neighboring_realm_same_rank_owner = {
			limit = {
				NOT = { this = scope:seizing_ruler }
			}
			root = {
				set_liege = prev
			}
		}

		add_stress = -20

		ai_chance = {
			base = 30

			modifier = {
				add = 30
				has_trait = ambitious
			}
		}
	}
}
