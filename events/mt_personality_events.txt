# 0001: Caught merchant cheating you
# Tests honesty - how do you respond to fraud?
mt_personality_events.1 = {
	type = character_event
	title = mt_personality_events.1.t
	desc = mt_personality_events.1.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = anger
	}

	right_portrait = {
		character = scope:cheating_merchant
		animation = worry
	}

	trigger = {
		# Must be a merchant
		OR = {
			has_trait = wandering_merchant
			has_trait = trade_post_owner
			has_trait = trading_company_head
			has_trait = merchant_prince
		}

		# Not already committed to honesty/dishonesty path
		NOT = { has_trait = trustworthy_dealer }
		NOT = { has_trait = dishonest_trader }
	}

	immediate = {
		# Create the cheating merchant
		create_character = {
			template = mt_merchant_templates
			save_scope_as = cheating_merchant
		}
	}

	# Option A: Report them (trustworthy_dealer path)
	option = {
		name = mt_personality_events.1.a

		# Gain reputation for honesty
		change_variable = {
			name = mt_reputation
			add = 10
		}

		# Track honest behavior
		mt_track_honest_dealing_effect = yes

		# Lose the potential profit from exploit
		add_gold = -50

		# Gain prestige for doing right thing
		add_prestige = 50

		# Chance to gain trustworthy_dealer trait
		random = {
			chance = 40
			mt_gain_merchant_trait_effect = { trait = trustworthy_dealer }
		}

		# Opinion boost from honest merchants
		every_character_with_trait = trustworthy_dealer = {
			limit = { knows_of_character = root }
			add_opinion = {
				target = root
				modifier = mt_honest_merchant_opinion
				opinion = 20
			}
		}

		ai_chance = {
			base = 40

			# More likely if already has high reputation
			modifier = {
				add = 30
				has_variable = mt_reputation
				var:mt_reputation >= 50
			}

			# Less likely if struggling financially
			modifier = {
				add = -20
				gold < 200
			}
		}

		stress_impact = {
			greedy = minor_stress_impact_gain
			ambitious = minor_stress_impact_gain
		}
	}

	# Option B: Exploit for profit (dishonest_trader path)
	option = {
		name = mt_personality_events.1.b

		# Gain gold from blackmail/exploitation
		add_gold = 200

		# Lose reputation
		change_variable = {
			name = mt_reputation
			add = -15
		}

		# Track dishonest behavior
		mt_track_dishonest_dealing_effect = yes

		# Chance to gain dishonest_trader trait
		random = {
			chance = 40
			mt_gain_merchant_trait_effect = { trait = dishonest_trader }
		}

		# Opinion penalty from honest merchants
		every_character_with_trait = trustworthy_dealer = {
			limit = { knows_of_character = root }
			add_opinion = {
				target = root
				modifier = mt_dishonest_merchant_opinion
				opinion = -30
			}
		}

		ai_chance = {
			base = 30

			# More likely if greedy or struggling
			modifier = {
				add = 40
				gold < 100
			}

			modifier = {
				add = 20
				has_trait = greedy
			}
		}

		stress_impact = {
			just = major_stress_impact_gain
			honest = major_stress_impact_gain
		}
	}

	# Option C: Negotiate fair solution (shrewd_negotiator path)
	option = {
		name = mt_personality_events.1.c

		# Balanced outcome
		add_gold = 75

		# Small reputation gain
		change_variable = {
			name = mt_reputation
			add = 5
		}

		# Track successful negotiation
		change_variable = {
			name = mt_successful_deals
			add = 1
		}

		# Chance to gain shrewd_negotiator trait
		random = {
			chance = 30
			mt_gain_merchant_trait_effect = { trait = shrewd_negotiator }
		}

		ai_chance = {
			base = 30

			# More likely if rational
			modifier = {
				add = 30
				ai_rationality >= 0
			}
		}
	}
}

# 0002: Risky venture with huge potential
# Tests risk appetite
mt_personality_events.2 = {
	type = character_event
	title = mt_personality_events.2.t
	desc = mt_personality_events.2.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = personality_rational
	}

	trigger = {
		# Must be a merchant
		OR = {
			has_trait = wandering_merchant
			has_trait = trade_post_owner
			has_trait = trading_company_head
			has_trait = merchant_prince
		}

		# Not already on risk path
		NOT = { has_trait = risk_taker }
		NOT = { has_trait = patient_investor }
		NOT = { has_trait = reckless_gambler }

		# Has enough gold to invest
		gold >= 300
	}

	# Option A: All in (risk_taker path)
	option = {
		name = mt_personality_events.2.a

		# Large investment
		remove_gold = 300

		# Track risky behavior
		mt_track_risky_venture_effect = yes

		# Random outcome
		random_list = {
			# 40% - huge success
			40 = {
				desc = mt_personality_events.2.a.success

				add_gold = 900
				add_prestige = 100

				# Likely to gain risk_taker
				random = {
					chance = 60
					mt_gain_merchant_trait_effect = { trait = risk_taker }
				}
			}

			# 30% - moderate success
			30 = {
				desc = mt_personality_events.2.a.moderate

				add_gold = 450
				add_prestige = 50
			}

			# 30% - total failure
			30 = {
				desc = mt_personality_events.2.a.failure

				# Lost all investment
				add_prestige = -50

				# Might become reckless_gambler if this happens often
				if = {
					limit = {
						has_variable = mt_risky_decisions
						var:mt_risky_decisions >= 3
					}
					random = {
						chance = 40
						mt_gain_merchant_trait_effect = { trait = reckless_gambler }
					}
				}
			}
		}

		ai_chance = {
			base = 20

			modifier = {
				add = 40
				ai_boldness >= 25
			}

			modifier = {
				add = 30
				ai_greed >= 25
			}

			modifier = {
				add = -30
				ai_rationality >= 25
			}
		}

		stress_impact = {
			craven = major_stress_impact_gain
			patient = minor_stress_impact_gain
		}
	}

	# Option B: Calculated portion (balanced path)
	option = {
		name = mt_personality_events.2.b

		# Moderate investment
		remove_gold = 150

		# Balanced outcome
		random_list = {
			60 = {
				desc = mt_personality_events.2.b.success
				add_gold = 300
				add_prestige = 50
			}

			40 = {
				desc = mt_personality_events.2.b.failure
				# Only lose half
				add_gold = 75
				add_prestige = -20
			}
		}

		ai_chance = {
			base = 50

			modifier = {
				add = 30
				ai_rationality >= 0
			}
		}
	}

	# Option C: Pass on it (patient_investor path)
	option = {
		name = mt_personality_events.2.c

		# No risk, no reward
		add_prestige = 25

		# Track conservative behavior
		add_character_flag = {
			flag = mt_prefers_long_term
			years = 5
		}

		# Chance to gain patient_investor
		random = {
			chance = 40
			mt_gain_merchant_trait_effect = { trait = patient_investor }
		}

		ai_chance = {
			base = 30

			modifier = {
				add = 40
				ai_rationality >= 25
			}

			modifier = {
				add = -30
				ai_greed >= 25
			}
		}

		stress_impact = {
			ambitious = minor_stress_impact_gain
			greedy = minor_stress_impact_gain
		}
	}
}

			}

			modifier = {
				add = -30
				ai_greed >= 25
			}
		}

		stress_impact = {
			greedy = major_stress_impact_gain
			callous = minor_stress_impact_gain
		}
	}

	# Option B: Loan with interest (practical path)
	option = {
		name = mt_personality_events.3.b

		# Give loan
		remove_gold = 100

		# Will get it back with interest eventually
		trigger_event = {
			id = mt_personality_events.101
			days = { 365 730 }
		}

		# Moderate opinion
		scope:poor_merchant = {
			add_opinion = {
				target = root

				opinion = 30
			}
		}

		ai_chance = {
			base = 50

			modifier = {
				add = 30
				ai_rationality >= 0
			}
		}
	}

	# Option C: Refuse (penny_pincher path)
	option = {
		name = mt_personality_events.3.c

		# Keep your gold
		# But lose reputation
		change_variable = {
			name = mt_reputation
			add = -10
		}

		# Track frugality
		change_variable = {
			name = mt_refused_expenses
			add = 1
		}
		add_character_flag = {
			flag = mt_extremely_frugal
			years = 5
		}

		# Negative opinion
		scope:poor_merchant = {
			add_opinion = {
				target = root

		}

		}

		ai_chance = {
			base = 25

			modifier = {
				add = 50
				ai_greed >= 25
			}

			modifier = {
				add = -30
				ai_compassion >= 0
			}
		}

		stress_impact = {
			compassionate = major_stress_impact_gain
			generous = major_stress_impact_gain
		}
	}
