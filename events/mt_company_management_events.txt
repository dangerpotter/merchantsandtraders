# 0005: Family member asks for job - nepotism dilemma
mt_company_management_events.5 = {
	type = character_event
	title = mt_company_management_events.5.t

	theme = family

	trigger = {
		has_trait = mt_merchant
		gold >= 60

		any_close_family_member = {
			is_adult = yes
			is_imprisoned = no
		}
	}

	left_portrait = {
		character = root

	}

	right_portrait = {
		character = scope:family_applicant
		animation = personality_content
	}

	immediate = {
		random_close_family_member = {
			limit = {
				is_adult = yes
				is_imprisoned = no
			}

		}

	# Hire them
	option = {
		name = mt_company_management_events.5.a

		hire_employee_effect = {
			role = flag:accountant
			quality = flag:experienced
		}

		scope:family_applicant = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 30
			}
		}

		stress_impact = {
			just = minor_stress_impact_gain
		}
	}

	# Politely refuse
	option = {
		name = mt_company_management_events.5.b

		scope:family_applicant = {
			add_opinion = {
				target = root

	}

# 0016: Employee asks for raise - deserve it or not?
mt_company_management_events.16 = {
	type = character_event
	title = mt_company_management_events.16.t
	desc = mt_company_management_events.16.desc
	theme = stewardship

	trigger = {
		has_trait = mt_merchant

		any_courtier = {
			has_variable = employer
			var:employer = root
			has_variable = employee_loyalty
			var:employee_loyalty >= 60
		}
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:requesting_employee
		animation = personality_bold
	}

	immediate = {
		random_courtier = {
			limit = {
				has_variable = employer
				var:employer = root
				has_variable = employee_loyalty
				var:employee_loyalty >= 60
			}
			save_scope_as = requesting_employee
		}
	}

	# Grant the raise
	option = {
		name = mt_company_management_events.16.a

		scope:requesting_employee = {
			manage_employee_loyalty_effect = {
				action = flag:pay_raise
			}
		}
