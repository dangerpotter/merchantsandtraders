# 0001: Plague reaches major trading city
mt_hist.0001 = {
	type = character_event
	title = mt_hist.0001.t
	desc = mt_hist.0001.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = fear
	}

	trigger = {
		OR = {
			has_trait = wandering_merchant
			has_trait = trade_post_owner
			has_trait = trading_company_head
		}
		year >= 867
		year <= 1453
		# More likely during actual Black Death period
		NOT = { has_character_modifier = mt_plague_survivor }
		NOT = { has_character_modifier = mt_plague_quarantine }
	}

	weight_multiplier = {
		base = 1
		modifier = {
			add = 3
			year >= 1346
			year <= 1353
		}
		modifier = {
			add = 2
			capital_province = {
				OR = {
					geographical_region = world_europe_west_italy
					geographical_region = world_europe_south_east
					geographical_region = custom_middle_east
				}
			}
		}
	}

	mean_time_to_happen = {
		months = 60

		modifier = {
			factor = 0.3
			year >= 1346
			year <= 1353
		}
	}

	immediate = {
		# Record plague exposure
		set_variable = {
			name = mt_plague_exposed
			value = yes
			days = 365
		}
	}

	# Continue trading - risky but profitable
	option = {
		name = mt_hist.0001.a

		custom_tooltip = mt_hist.0001.a.tt

		# High risk, high reward
		random = {
			chance = 40
			# Contract plague
			add_trait = ill
			add_stress = 50
			trigger_event = {
				id = mt_hist.0002
				days = { 7 14 }
			}
		}

		random = {
			chance = 60
			# Survive and profit enormously
			add_gold = 200
			add_prestige = 50
			add_character_modifier = {
				modifier = mt_plague_profiteer
				months = 36
			}
		}

		ai_chance = {
			base = 30
			modifier = {
				add = 30
				has_trait = greedy
			}
			modifier = {
				add = 20
				has_trait = brave
			}
			modifier = {
				add = -40
				has_trait = craven
			}
		}
	}

	# Cease operations - safe but no income
	option = {
		name = mt_hist.0001.b

		custom_tooltip = mt_hist.0001.b.tt

		add_character_modifier = {
			modifier = mt_plague_quarantine
			months = 12
		}

		# Lost opportunities
		remove_short_term_gold = 50
		add_stress = 20

		# But safe
		add_trait = calm

		ai_chance = {
			base = 60
			modifier = {
				add = 30
				has_trait = craven
			}
			modifier = {
				add = 20
				has_trait = cautious_leader
			}
		}
	}

	# Trade in essential goods only
	option = {
		name = mt_hist.0001.c

		trigger = {
			gold >= 100
		}

		custom_tooltip = mt_hist.0001.c.tt

		remove_short_term_gold = 100

		# Medium risk, good reward
		random = {
			chance = 25
			add_trait = ill
			add_stress = 30
		}

		random = {
			chance = 75
			# Profit from medicine and grain
			add_gold = 300
			add_prestige = 75
			add_piety = 25
			add_character_modifier = {
				modifier = mt_essential_goods_trader
				months = 24
			}
		}

		ai_chance = {
			base = 40
			modifier = {
				add = 30
				gold >= 200
			}
		}
	}
}
