# 0001: First successful trade - gain confidence
mt_career.0001 = {
	type = character_event
	title = mt_career.0001.t
	desc = mt_career.0001.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = happiness
	}

	trigger = {
		has_trait = wandering_merchant
		NOT = { has_character_flag = had_first_trade_success }
		gold >= 50
	}

	weight_multiplier = {
		base = 1

		# More likely early in career
		modifier = {
			add = 2
			trait_duration = {
				trait = wandering_merchant
				months <= 12
			}
		}
	}

	immediate = {
		add_character_flag = had_first_trade_success
	}

	# Celebrate your success
	option = {
		name = mt_career.0001.a

		add_prestige = 50
		add_gold = 25
		add_stress = -10

		# Chance to gain gregarious trait
		random = {
			chance = 15
			add_trait = gregarious
		}

		ai_chance = {
			base = 60

			modifier = {
				add = 30
				has_trait = gregarious
			}
		}
	}

	# Stay humble and focused
	option = {
		name = mt_career.0001.b

		add_gold = 50
		add_learning = 1

		# Chance to gain diligent trait
		random = {
			chance = 20
			add_trait = diligent
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				has_trait = humble
			}
		}
	}
}

# 0003: Lost all your money
mt_career.0003 = {
	type = character_event
	title = mt_career.0003.t
	desc = mt_career.0003.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = sadness
	}

	trigger = {
		has_trait = wandering_merchant
		gold <= 10
		NOT = { has_character_flag = experienced_bankruptcy }
	}

	weight_multiplier = {
		base = 1

		# Less likely if you have financial focus
		modifier = {
			factor = 0.3
			stewardship >= 12
		}
	}

	immediate = {
		add_character_flag = experienced_bankruptcy
	}

	# Take a loan from fellow merchants
	option = {
		name = mt_career.0003.a

		trigger = {
			any_courtier = {
				OR = {
					has_trait = trade_post_owner
					has_trait = trading_company_head
					has_trait = merchant_prince
				}
			}
		}

		add_gold = 100

		add_character_modifier = {
			modifier = mt_merchant_debt
			years = 5
		}

		add_stress = 20

		ai_chance = {
			base = 50

			modifier = {
				add = 30
				has_trait = ambitious
			}
		}
	}

	# Start over from scratch
	option = {
		name = mt_career.0003.b

		add_gold = 25
		add_stress = 30

		# Chance to gain resilient or stubborn trait
		random = {
			chance = 25
			random_list = {
				50 = { add_trait = stubborn }
				50 = { add_trait = ambitious }
			}
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				has_trait = diligent
			}
		}
	}

	# Give up merchant life (only if desperate)
	option = {
		name = mt_career.0003.c

		trigger = {
			OR = {
				has_trait = craven
				has_trait = lazy
				age >= 50
			}
		}

		remove_trait = wandering_merchant
		add_prestige = -100
		add_stress = -20

		# Severe opinion penalty from other merchants
		every_courtier = {
			limit = {
				OR = {
					has_trait = wandering_merchant
					has_trait = trade_post_owner
					has_trait = trading_company_head
					has_trait = merchant_prince
				}
			}
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -30
			}
		}

		ai_chance = {
			base = 10

			modifier = {
				add = 30
				has_trait = craven
			}

			modifier = {
				add = 20
				age >= 55
			}
		}
	}
}

# 0006: Accused of fraud - defend reputation or pay damages
mt_career.0006 = {
	type = character_event
	title = mt_career.0006.t
	desc = mt_career.0006.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = worry
	}

	right_portrait = {
		character = scope:accuser
		animation = anger
	}

	trigger = {
		OR = {
			has_trait = wandering_merchant
			has_trait = trade_post_owner
			has_trait = trading_company_head
		}
		any_courtier = {
			NOT = { this = root }
			is_adult = yes
		}
	}

	weight_multiplier = {
		base = 1

		# More likely if you have rivals
		modifier = {
			add = 2
			any_relation = {
				type = rival
				is_adult = yes
			}
		}
	}

	immediate = {
		if = {
			limit = {
				any_relation = {
					type = rival
					is_adult = yes
				}
			}
			random_relation = {
				type = rival
				limit = { is_adult = yes }
				save_scope_as = accuser
			}
		}
		else = {
			random_courtier = {
				limit = {
					NOT = { this = root }
					is_adult = yes
				}
				save_scope_as = accuser
			}
		}
	}

	# Defend your reputation publicly
	option = {
		name = mt_career.0006.a

		trigger = {
			diplomacy >= 12
		}

		duel = {
			skill = diplomacy
			target = scope:accuser

			50 = { # Success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
				}
				desc = mt_career.0006.a.success

				send_interface_toast = {
					title = mt_career.0006.a.success
					left_icon = root

					add_prestige = 100

					scope:accuser = {
						add_opinion = {
							target = root
							modifier = humiliated_opinion
							opinion = -50
						}
					}

					# Gain respect from peers
					every_courtier = {
						limit = {
							OR = {
								has_trait = wandering_merchant
								has_trait = trade_post_owner
								has_trait = trading_company_head
								has_trait = merchant_prince
							}
						}
						add_opinion = {
							target = root
							modifier = respect_opinion
							opinion = 10
						}
					}
				}
			}

			50 = { # Failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3
				}
				desc = mt_career.0006.a.failure

				send_interface_toast = {
					title = mt_career.0006.a.failure
					left_icon = root

					add_prestige = -150
					add_stress = 40

					add_character_modifier = {
						modifier = mt_damaged_reputation
						years = 5
					}
				}
			}
		}

		ai_chance = {
			base = 50

			modifier = {
				add = 40
				diplomacy >= 14
			}

			modifier = {
				add = 20
				has_trait = brave
			}
		}
	}

	# Pay damages to avoid scandal
	option = {
		name = mt_career.0006.b

		trigger = {
			gold >= 150
		}

		remove_short_term_gold = 150
		add_prestige = -50

		scope:accuser = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				gold >= 500
			}

			modifier = {
				add = 20
				has_trait = craven
			}
		}
	}

	# Counter-accuse them
	option = {
		name = mt_career.0006.c

		trigger = {
			intrigue >= 10
		}

		duel = {
			skill = intrigue
			target = scope:accuser

			50 = { # Success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
				}
				desc = mt_career.0006.c.success

				send_interface_toast = {
					title = mt_career.0006.c.success
					left_icon = root

					scope:accuser = {
						add_prestige = -100
						add_stress = 30

						add_opinion = {
							target = root
							modifier = rival_opinion
							opinion = -75
						}
					}

					add_prestige = 75
				}
			}

			50 = { # Failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3
				}
				desc = mt_career.0006.c.failure

				send_interface_toast = {
					title = mt_career.0006.c.failure
					left_icon = root

					add_prestige = -200
					remove_short_term_gold = 100
					add_stress = 50

					add_character_modifier = {
						modifier = mt_damaged_reputation
						years = 10
					}
				}
			}
		}

		ai_chance = {
			base = 30

			modifier = {
				add = 40
				intrigue >= 14
			}

			modifier = {
				add = 30
				has_trait = vengeful
			}
		}
	}
}

# 0023: Retirement consideration - pass business to heir
mt_career.0023 = {
	type = character_event
	title = mt_career.0023.t
	desc = mt_career.0023.desc
	theme = family

	left_portrait = {
		character = root
		animation = personality_content
	}

	right_portrait = {
		character = scope:heir
		animation = personality_rational
	}

	trigger = {
		OR = {
			has_trait = trading_company_head
			has_trait = merchant_prince
		}
		age >= 60
		any_child = {
			is_adult = yes
			OR = {
				has_trait = wandering_merchant
				has_trait = trade_post_owner
				stewardship >= 10
			}
		}
		NOT = { has_character_flag = considered_retirement }
	}

	weight_multiplier = {
		base = 1

		modifier = {
			add = 3
			age >= 70
		}
	}

	immediate = {
		add_character_flag = considered_retirement

		random_child = {
			limit = {
				is_adult = yes
				OR = {
					has_trait = wandering_merchant
					has_trait = trade_post_owner
					stewardship >= 10
				}
			}
			save_scope_as = heir
		}
	}

	option = {
		name = mt_career.0023.a

					has_trait = wandering_merchant

		add_piety = 200

		add_character_modifier = {
			modifier = mt_retired_merchant
			years = 20
		}

		ai_chance = {
			base = 50

			modifier = {
				add = 40
				age >= 70
			}

			modifier = {
				add = 30
				has_trait = content
			}
		}
	}

	# Share the business but stay involved
	option = {
		name = mt_career.0023.b

		scope:heir = {
			add_gold = 500
			add_prestige = 250
			add_diplomacy = 1
			add_stewardship = 1

				target = root
				modifier = grateful_opinion
				opinion = 50
			}
		}

		add_stress = -20

		add_character_modifier = {
			modifier = mt_mentoring_heir
			years = 15
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				has_trait = diligent
			}
		}
	}

	# Keep working - not ready to retire
	option = {
		name = mt_career.0023.c

		add_prestige = 100
		add_stress = 20

		scope:heir = {
			add_opinion = {
				target = root
				modifier = disappointed_opinion

			}
		}

				has_trait = ambitious
			}
