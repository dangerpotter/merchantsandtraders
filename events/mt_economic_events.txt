# 0001: Silk market crashes - oversupply causes prices to drop 60%
mt_econ.0001 = {
	type = character_event
	title = mt_econ.0001.t
	desc = mt_econ.0001.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = shock
	}

	trigger = {
		has_trait = wandering_merchant
		gold >= 20
	}

	mean_time_to_happen = {
		months = 36

		modifier = {
			factor = 0.7
			any_held_title = {
				tier = tier_county
				county = {
					has_county_modifier = mt_trade_good_silk
				}
			}
		}
	}

	immediate = {
		# Store whether character has silk goods
		if = {
			limit = {
				any_held_title = {
					tier = tier_county
					county = {
						has_county_modifier = mt_trade_good_silk
					}
				}
			}
			set_variable = {
				name = mt_has_silk_goods
				value = yes
				days = 30
			}
		}
	}

	# Hold the goods - wait for recovery
	option = {
		name = mt_econ.0001.a

		trigger = {
			has_variable = mt_has_silk_goods
		}

		# Immediate loss from holding devalued goods
		remove_short_term_gold = 15
		add_stress = 20

		# But may recover in future
		trigger_event = {
			id = mt_econ.0026
			days = { 180 365 }
		}

		ai_chance = {
			base = 40
			modifier = {
				add = 30
				has_trait = patient
			}
			modifier = {
				add = 20
				gold >= 100
			}
		}
	}

	# Sell at loss - cut your losses
	option = {
		name = mt_econ.0001.b

		trigger = {
			has_variable = mt_has_silk_goods
		}

		# Sell at 40% of value
		remove_short_term_gold = 25
		add_stress = 30

		# Stress relief from accepting reality
		add_prestige = -20

		ai_chance = {
			base = 40
			modifier = {
				add = 30
				gold < 50
			}
		}
	}

	# Switch to different goods
	option = {
		name = mt_econ.0001.c

		# Investment in new goods
		trigger = {
			gold >= 30
		}

		remove_short_term_gold = 30

		# Diversification pays off
		random = {
			chance = 60
			add_gold = 45
			custom_tooltip = mt_econ.0001.c.success
		}

		random = {
			chance = 40
			add_stress = 15
			custom_tooltip = mt_econ.0001.c.failure
		}

		ai_chance = {
			base = 50
			modifier = {
				add = 30
				stewardship >= 12
			}
		}
	}

	# Don't trade in silk - just observe
	option = {
		name = mt_econ.0001.d

		trigger = {
			NOT = { has_variable = mt_has_silk_goods }
		}

		# Learn from others' mistakes
		add_stewardship_lifestyle_xp = 50
		add_stress = -5

		custom_tooltip = mt_econ.0001.d.tt

		ai_chance = {
			base = 100
		}
	}
}

# 0002: Grain shortage - prices surge 200%
mt_econ.0002 = {
	type = character_event
	title = mt_econ.0002.t
	desc = mt_econ.0002.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = worry
	}

	trigger = {
		has_trait = wandering_merchant
	}

	mean_time_to_happen = {
		months = 30

		modifier = {
			factor = 0.5
			any_character = {
				is_at_war = yes
			}
		}

		modifier = {
			factor = 2.0
			current_month >= 9
			current_month <= 11
		}
	}

	immediate = {
		# Check if character has grain
		if = {
			limit = {
				any_held_title = {
					tier = tier_county
					county = {
						has_county_modifier = mt_trade_good_grain
					}
				}
			}
			set_variable = {
				name = mt_has_grain_stockpile
				value = yes
				days = 30
			}
		}
	}

	# Sell grain stockpile at massive profit
	option = {
		name = mt_econ.0002.a

		trigger = {
			has_variable = mt_has_grain_stockpile
		}

		add_gold = 60
		add_prestige = 40

		# But people are starving
		add_piety = -30
		add_stress = 10

		if = {
			limit = { has_trait = greedy }
			add_stress = -10
		}

		ai_chance = {
			base = 50
			modifier = {
				add = 40
				has_trait = greedy
			}
			modifier = {
				add = -40
				has_trait = charitable
			}
		}
	}

	# Sell at fair prices to help people
	option = {
		name = mt_econ.0002.b

		trigger = {
			has_variable = mt_has_grain_stockpile
		}

		add_gold = 25
		add_piety = 50
		add_prestige = 30

		if = {
			limit = { has_variable = mt_reputation }
			change_variable = {
				name = mt_reputation
				add = 15
			}
		}

		ai_chance = {
			base = 30
			modifier = {
				add = 50
				has_trait = charitable
			}
			modifier = {
				add = 30
				has_trait = compassionate
			}
		}
	}

	# Buy grain to stockpile for higher prices
	option = {
		name = mt_econ.0002.c

		trigger = {
			gold >= 40
			NOT = { has_variable = mt_has_grain_stockpile }
		}

		remove_short_term_gold = 40

		# Gamble on even higher prices
		random_list = {
			40 = {
				# Prices go even higher
				add_gold = 80
				add_prestige = 25
				custom_tooltip = mt_econ.0002.c.success
			}
			60 = {
				# Market stabilizes
				add_gold = 35
				custom_tooltip = mt_econ.0002.c.neutral
			}
		}

		ai_chance = {
			base = 40
			modifier = {
				add = 30
				has_trait = ambitious
			}
		}
	}

	# No grain - just suffer high food costs
	option = {
		name = mt_econ.0002.d

		trigger = {
			NOT = { has_variable = mt_has_grain_stockpile }
		}

		remove_short_term_gold = 10
		add_stress = 15

		custom_tooltip = mt_econ.0002.d.tt

		ai_chance = {
			base = 100
		}
	}
}
