# 0001 - Disgruntled employee offers to spy
mt_espionage.0001 = {
	type = character_event
	title = mt_espionage.0001.t
	desc = mt_espionage.0001.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = worry
	}

	trigger = {
		is_adult = yes
		NOT = { has_variable = recently_offered_spy }
		any_known_character = {
			is_adult = yes
			NOT = { this = root }
			is_alive = yes
		}
	}

	immediate = {
		set_variable = {
			name = recently_offered_spy
			days = 180
		}
		random_known_character = {
			limit = {
				is_adult = yes
				NOT = { this = root }
			}
			save_scope_as = rival_merchant
		}
	}

	# Accept the offer - risky but cheap
	option = {
		name = mt_espionage.0001.a

		# Low quality spy but loyal employee knows things
		set_variable = {
			name = spy_quality
			value = 1
		}
		set_variable = {
			name = spy_loyalty
			value = 75
		}
		set_variable = {
			name = has_active_spy
			value = yes
		}

		add_gold = -30

		# Small intrigue boost
		add_intrigue_skill = 1

		# But risk the employee is actually a plant
		random = {
			chance = 20
			set_variable = {
				name = spy_is_double_agent
				value = yes
			}
			custom_tooltip = mt_espionage.0001.a.risk
		}

		ai_chance = {
			base = 40
			modifier = {
				factor = 2
				intrigue >= 12
			}
			modifier = {
				factor = 0.5
				has_trait = trusting
			}
		}
	}

	# Report to the rival - gain honor
	option = {
		name = mt_espionage.0001.b
		trait = honest

		add_prestige = 50

		scope:rival_merchant = {
			add_opinion = {
				target = root
				modifier = warned_of_spy_opinion
				opinion = 40
			}
		}

		stress_impact = {
			deceitful = minor_stress_impact_gain
			schemer = minor_stress_impact_gain
		}

		ai_chance = {
			base = 30
			modifier = {
				factor = 3
				has_trait = honest
			}
			modifier = {
				factor = 2
				has_trait = just
			}
		}
	}

	# Refuse - stay out of it
	option = {
		name = mt_espionage.0001.c

		add_prestige = 10

		stress_impact = {
			ambitious = minor_stress_impact_gain
		}

		ai_chance = {
			base = 30
			modifier = {
				factor = 2
				has_trait = content
			}
		}
	}
}

# 0002 - Professional spy seeks employment
mt_espionage.0002 = {
	type = character_event
	title = mt_espionage.0002.t
	desc = mt_espionage.0002.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = scheme
	}

	trigger = {
		is_adult = yes
		gold >= 100
		NOT = { has_variable = professional_spy_offered }
	}

	immediate = {
		set_variable = {
			name = professional_spy_offered
			days = 365
		}
	}

	# Hire the professional - expensive but skilled
	option = {
		name = mt_espionage.0002.a

		remove_short_term_gold = 100

		# Expert quality spy
		set_variable = {
			name = spy_quality
			value = 3
		}
		set_variable = {
			name = spy_loyalty
			value = 60
		}
		set_variable = {
			name = spy_success_chance
			value = 85
		}
		set_variable = {
			name = has_active_spy
			value = yes
		}

		add_prestige = 25

		custom_tooltip = mt_espionage.0002.a.effect

		ai_chance = {
			base = 60
			modifier = {
				factor = 2
				gold >= 300
			}
			modifier = {
				factor = 2
				intrigue >= 15
			}
			modifier = {
				factor = 0.3
				gold < 150
			}
		}
	}

	# Negotiate lower price - medium quality
	option = {
		name = mt_espionage.0002.b
		skill = diplomacy
		duel = {
			skill = diplomacy
			value = 14
		}

		remove_short_term_gold = 60

		# Competent spy at discount
		if = {
			limit = { scope:duel_won = yes }
			set_variable = {
				name = spy_quality
				value = 2
			}
			set_variable = {
				name = spy_loyalty
				value = 55
			}
			custom_tooltip = mt_espionage.0002.b.success
		}
		else = {
			# Amateur spy, insulted by negotiation
			set_variable = {
				name = spy_quality
				value = 1
			}
			set_variable = {
				name = spy_loyalty
				value = 40
			}
			custom_tooltip = mt_espionage.0002.b.failure
		}

		set_variable = {
			name = has_active_spy
			value = yes
		}

		ai_chance = {
			base = 30
			modifier = {
				factor = 3
				diplomacy >= 14
			}
			modifier = {
				factor = 2
				gold < 150
			}
		}
	}

	# Decline - too expensive
	option = {
		name = mt_espionage.0002.c

		add_prestige = 5

		ai_chance = {
			base = 10
			modifier = {
				factor = 5
				gold < 100
			}
		}
	}
}

# 0004 - Rival's employee approaches you
mt_espionage.0004 = {
	type = character_event
	title = mt_espionage.0004.t
	desc = mt_espionage.0004.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = worry
	}

	right_portrait = {
		character = scope:rival_target
		animation = anger
	}

	trigger = {
		is_adult = yes
		gold >= 50
		any_relation = {
			type = rival
			is_alive = yes
		}
		NOT = { has_variable = rival_employee_offer }
	}

	immediate = {
		set_variable = {
			name = rival_employee_offer
			days = 180
		}
		random_relation = {
			type = rival
			save_scope_as = rival_target
		}
	}

	# Accept - perfect inside access
	option = {
		name = mt_espionage.0004.a

		remove_short_term_gold = 50

		# Good chance this is genuine
		random_list = {
			70 = {
				# Legitimate offer - competent insider
				set_variable = {
					name = spy_quality
					value = 2
				}
				set_variable = {
					name = spy_loyalty
					value = 70
				}
				set_variable = {
					name = has_active_spy
					value = yes
				}
				set_variable = {
					name = spy_has_inside_access
					value = scope:rival_target
				}

				custom_tooltip = mt_espionage.0004.a.success
			}
			30 = {
				# Trap! Double agent
				set_variable = {
					name = spy_quality
					value = 1
				}
				set_variable = {
					name = spy_loyalty
					value = 20
				}
				set_variable = {
					name = spy_is_double_agent
					value = yes
				}
				set_variable = {
					name = has_active_spy
					value = yes
				}

				scope:rival_target = {
					add_opinion = {
						target = root
						modifier = fell_for_trap_opinion
						opinion = -30
					}
				}

				custom_tooltip = mt_espionage.0004.a.trap
			}
		}

		ai_chance = {
			base = 50
			modifier = {
				factor = 2
				intrigue >= 14
			}
			modifier = {
				factor = 0.5
				has_trait = trusting
			}
			modifier = {
				factor = 2
				has_trait = paranoid
			}
		}
	}

	# Test their loyalty first - costs less but delays
	option = {
		name = mt_espionage.0004.b
		trait = paranoid

		remove_short_term_gold = 25

		# Run background check
		random = {
			chance = 60
			modifier = {
				add = 20
				intrigue >= 15
			}

			# Discovered it's a trap!
			add_prestige = 100
			custom_tooltip = mt_espionage.0004.b.discovered

			scope:rival_target = {
				add_opinion = {
					target = root
					modifier = foiled_my_plan_opinion
					opinion = -50
				}
			}
		}

		# If not discovered, proceed cautiously
		if = {
			limit = {
				NOT = { has_variable = discovered_trap }
			}
			trigger_event = {
				id = mt_espionage.0004
				days = 30
			}
		}

		stress_impact = {
			trusting = minor_stress_impact_gain
		}

		ai_chance = {
			base = 30
			modifier = {
				factor = 3
				has_trait = paranoid
			}
			modifier = {
				factor = 2
				intrigue >= 16
			}
		}
	}

	# Decline - too suspicious
	option = {
		name = mt_espionage.0004.c

		add_prestige = 20

		# Maybe inform your rival
		random = {
			chance = 30
			scope:rival_target = {
				add_opinion = {
					target = root
					modifier = honorable_conduct_opinion
					opinion = 20
				}
			}
			custom_tooltip = mt_espionage.0004.c.inform
		}

		ai_chance = {
			base = 20
			modifier = {
				factor = 3
				has_trait = honest
			}
		}
	}
}

	}

	}

	immediate = {
		random_known_character = {

				NOT = { this = root }
			}
			save_scope_as = secret_buyer
		}
	}

	# Launch full investigation
	option = {
		name = mt_espionage.0040.a
		trait = vengeful

		remove_short_term_gold = 200

		# Find the mole
		random = {
			chance = 70
			modifier = {
				add = 20
				intrigue >= 16
			}

			# Found and punished
			add_prestige = 150
			custom_tooltip = mt_espionage.0040.a.success
		}

		stress_impact = {
			forgiving = minor_stress_impact_gain
		}

		ai_chance = {
			base = 50
			modifier = {
				factor = 3
				has_trait = vengeful
			}
		}
	}

	# Damage control - spin the story
	option = {
		name = mt_espionage.0040.b
		skill = diplomacy
		duel = {
			skill = diplomacy
			value = 15
		}

		remove_short_term_gold = 150

		if = {
			limit = { scope:duel_won = yes }
			# Successfully spun
			add_prestige = 75
			custom_tooltip = mt_espionage.0040.b.success
		}
		else = {
			# Failed to contain
			add_prestige = -100
			custom_tooltip = mt_espionage.0040.b.failure
		}

		ai_chance = {
			base = 30
			modifier = {
				factor = 3
				diplomacy >= 16
			}
		}
	}

	# Accept the loss
	option = {
		name = mt_espionage.0040.c

		add_prestige = -200

		# Major reputation hit
		add_character_modifier = {
			modifier = mt_major_security_breach
			years = 3
		}

		# Remove compromised secrets
		remove_variable = has_stolen_secrets

		add_stress = 75

		custom_tooltip = mt_espionage.0040.c.effect

		ai_chance = {
			base = 20
			modifier = {
				factor = 3
				gold < 200
			}
		}
	}
