# 0003: Robbed at seedy tavern
mt_flavor.0003 = {
	type = character_event

	desc = mt_flavor.0003.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = anger
	}

	trigger = {
		has_trait = wandering_merchant
		gold >= 10
	}

	mean_time_to_happen = {
		months = 24
	}

	# Pay up to avoid trouble
	option = {
		name = mt_flavor.0003.a

		remove_short_term_gold = 8
		add_stress = 10

		ai_chance = {
			base = 60
			modifier = {
				add = 30
				has_trait = craven
			}
		}
	}

	# Try to chase them down
	option = {
		name = mt_flavor.0003.b

		trigger = {
			prowess >= 8
		}

		random_list = {
			50 = {
				# Success - get money back plus extra
				add_gold = 15
				add_prestige = 15
				add_stress = -5
			}
			50 = {
				# Failure - lose more money and get hurt
				remove_short_term_gold = 12
				add_stress = 20
				add_health = -0.25
			}
		}

		ai_chance = {
			base = 40
			modifier = {
				add = 30
				has_trait = brave
			}
		}
	}

	# Accept the loss
	option = {
		name = mt_flavor.0003.c

		remove_short_term_gold = 8
		add_stress = 5

		if = {
			limit = { has_trait = calm }
			add_stress = -3
		}

		ai_chance = {
			base = 30
			modifier = {
				add = 40
				has_trait = calm
			}
		}
	}
}
