# 0003: Tier up to Respected Trader (50 reputation)
mt_reputation.0003 = {
	type = character_event
	title = mt_reputation.0003.t
	desc = mt_reputation.0003.desc
	theme = diplomacy

	left_portrait = {
		character = root
		animation = personality_bold
	}

	right_portrait = {
		character = scope:fellow_merchant
		animation = personality_honorable
	}

	trigger = {
		has_character_modifier = mt_reputation_respected_trader
		NOT = { has_character_flag = celebrated_respected_trader }
		any_courtier = {
			OR = {
				has_trait = trade_post_owner
				has_trait = trading_company_head
			}
		}
	}

	weight_multiplier = {
		base = 100
	}

	immediate = {
		add_character_flag = celebrated_respected_trader

		random_courtier = {
			limit = {
				OR = {
					has_trait = trade_post_owner
					has_trait = trading_company_head
				}
			}
			save_scope_as = fellow_merchant
		}
	}

	# Accept their congratulations proudly
	option = {
		name = mt_reputation.0003.a

		add_prestige = 150

		scope:fellow_merchant = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 25
			}
		}

		# Other merchants respect you
		every_courtier = {
			limit = {
				OR = {
					has_trait = wandering_merchant
					has_trait = trade_post_owner
					has_trait = trading_company_head
				}
			}
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 10
			}
		}

		custom_tooltip = mt_reputation.0003.a.tt

		ai_chance = {
			base = 60
		}
	}

	# Stay humble
	option = {
		name = mt_reputation.0003.b

		add_prestige = 100
		add_piety = 50

		scope:fellow_merchant = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
		}

		custom_tooltip = mt_reputation.0003.a.tt

		ai_chance = {
			base = 40
			modifier = {
				add = 40
				has_trait = humble
			}
		}
	}
}

# 0011: High reputation - offered royal contract
mt_reputation.0011 = {
	type = character_event
	title = mt_reputation.0011.t
	desc = mt_reputation.0011.desc
	theme = diplomacy

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:offering_ruler
		animation = personality_honorable
	}

	trigger = {
		reputation_above_trigger = { amount = 200 }
		exists = liege
		liege = {
			is_ruler = yes
		}
		NOT = { has_character_flag = offered_royal_contract }
	}

	weight_multiplier = {
		base = 1

		modifier = {
			add = 2
			reputation_above_trigger = { amount = 300 }
		}
	}

	immediate = {
		add_character_flag = offered_royal_contract

		liege = {
			save_scope_as = offering_ruler
		}
	}

	# Accept exclusive contract
	option = {
		name = mt_reputation.0011.a

		add_character_modifier = {
			modifier = mt_royal_contract
			years = 15
		}

		add_prestige = 200
		add_gold = 500

		scope:offering_ruler = {
			add_opinion = {
				target = root
				modifier = loyal_opinion
				opinion = 60
			}
		}

		ai_chance = {
			base = 60
		}
	}

	# Negotiate flexible terms
	option = {
		name = mt_reputation.0011.b

		trigger = {

			base = 30

				add = 40
				diplomacy >= 16
			}
		}
	}

	# Decline politely
	option = {
		name = mt_reputation.0011.c

		add_prestige = 100

			add_opinion = {
				target = root
				modifier = disappointed_opinion

			}
		}

		ai_chance = {
			base = 10
			modifier = {
				add = 40
				has_trait = proud
			}
		}
	}
}
