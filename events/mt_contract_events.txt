# 0001: Ruler wants exclusive grain supply - lucrative but risky
mt_contract_events.0001 = {
	type = character_event
	title = mt_contract_events.0001.t
	desc = mt_contract_events.0001.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:contract_buyer
		animation = personality_bold
	}

	trigger = {
		is_any_merchant_trigger = yes
		NOT = { has_variable = mt_supply_contract_active }
	}

	immediate = {
		# Find a suitable buyer (ruler)
		if = {
			limit = {
				exists = liege
				liege = {
					is_ruler = yes
					gold >= 500
				}
			}
			liege = {
				save_scope_as = contract_buyer
			}
		}
		else_if = {
			limit = {
				any_vassal_or_below = {
					is_ruler = yes
					gold >= 300
				}
			}
			random_vassal_or_below = {
				limit = {
					is_ruler = yes
					gold >= 300
				}
				save_scope_as = contract_buyer
			}
		}
		else = {
			# Create a temporary buyer scope
			save_scope_as = contract_buyer
		}
	}

	# Option A: Accept the lucrative terms (high risk, high reward)
	option = {
		name = mt_contract_events.0001.a
		flavor = mt_contract_events.0001.a.flavor

		# Set contract parameters
		set_variable = {
			name = contract_partner
			value = scope:contract_buyer
		}

		set_variable = {
			name = contract_duration
			value = 5
		}

		set_variable = {
			name = contract_price
			value = 100
		}

		set_variable = {
			name = contract_good
			value = flag:grain
		}

		# Create the supply contract
		create_supply_contract_effect = yes

		# Immediate payment
		add_gold = 200

		# High pressure modifier
		add_character_modifier = {
			modifier = mt_high_stakes_contract
			years = 5
		}

		# Stress if cautious
		if = {
			limit = { has_trait = cautious }
			add_stress = 20
		}

		# Reputation gain for landing major contract
		if = {
			limit = { has_variable = merchant_reputation }
			change_variable = {
				name = merchant_reputation
				add = 10
			}
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				has_trait = ambitious
			}

			modifier = {
				add = 20
				has_trait = greedy
			}

			modifier = {
				add = -40
				has_trait = cautious
			}

			modifier = {
				add = 20
				gold < 300
			}
		}
	}

	# Option B: Negotiate for safer terms (lower risk, lower reward)
	option = {
		name = mt_contract_events.0001.b

		# Set safer contract parameters
		set_variable = {
			name = contract_partner
			value = scope:contract_buyer
		}

		set_variable = {
			name = contract_duration
			value = 3
		}

		set_variable = {
			name = contract_price
			value = 60
		}

		set_variable = {
			name = contract_good
			value = flag:grain
		}

		# Create the supply contract
		create_supply_contract_effect = yes

		# Smaller immediate payment
		add_gold = 100

		# Reduce stress
		if = {
			limit = { stress > 0 }
			add_stress = -10
		}

		# Cautious approval
		if = {
			limit = { has_trait = cautious }
			add_prestige = 50
		}

		ai_chance = {
			base = 50

			modifier = {
				add = 40
				has_trait = cautious
			}

			modifier = {
				add = 20
				has_trait = content
			}

			modifier = {
				add = -30
				has_trait = ambitious
			}

			modifier = {
				add = 30
				gold >= 500
			}
		}
	}

	# Option C: Decline - too risky
	option = {
		name = mt_contract_events.0001.c

		# Small reputation loss for turning down opportunity
		if = {
			limit = { has_variable = merchant_reputation }
			change_variable = {
				name = merchant_reputation
				subtract = 5
			}
		}

		# Slight opinion loss with buyer
		reverse_add_opinion = {
			target = scope:contract_buyer
			modifier = disappointed_opinion
			opinion = -10
		}

		# But reduce stress
		add_stress = -15

		ai_chance = {
			base = 10

			modifier = {
				add = 40
				has_trait = craven
			}

			modifier = {
				add = 30
				stress >= 75
			}

			modifier = {
				add = -40
				has_trait = greedy
			}
		}
	}
}

# 0002: Merchant offers partnership contract
mt_contract_events.0002 = {
	type = character_event
	title = mt_contract_events.0002.t
	desc = mt_contract_events.0002.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:merchant_partner
		animation = personality_compassionate
	}

	trigger = {
		is_any_merchant_trigger = yes
		gold >= 200
	}

	immediate = {
		# Find a merchant partner
		if = {
			limit = {
				any_courtier = {
					is_any_merchant_trigger = yes
					NOT = { this = root }
				}
			}
			random_courtier = {
				limit = {
					is_any_merchant_trigger = yes
					NOT = { this = root }
				}
				save_scope_as = merchant_partner
			}
		}
		else = {
			save_scope_as = merchant_partner
		}
	}

	# Option A: Equal partnership
	option = {
		name = mt_contract_events.0002.a

		# Create mutual benefit contract
		set_variable = {
			name = contract_partner
			value = scope:merchant_partner
		}

		set_variable = {
			name = contract_duration
			value = 5
		}

		set_variable = {
			name = contract_price
			value = 50
		}

		create_supply_contract_effect = yes

		# Build strong relationship
		add_opinion = {
			target = scope:merchant_partner
			modifier = mt_trusted_partner_opinion
			opinion = 30
		}

		reverse_add_opinion = {
			target = scope:merchant_partner
			modifier = mt_trusted_partner_opinion
			opinion = 30
		}

		# Add partnership bonus
		add_character_modifier = {
			modifier = mt_merchant_partnership
			years = 5
		}

		add_prestige = 75

		ai_chance = {
			base = 60

			modifier = {
				add = 20
				has_trait = just
			}

			modifier = {
				add = 15
				has_trait = generous
			}
		}
	}

	# Option B: Dominant position (you lead)
	option = {
		name = mt_contract_events.0002.b

		trigger = {
			gold >= 300
			diplomacy >= 12
		}

		# Better terms for you
		set_variable = {
			name = contract_partner
			value = scope:merchant_partner
		}

		set_variable = {
			name = contract_duration
			value = 5
		}

		set_variable = {
			name = contract_price
			value = 75
		}

		create_exclusive_purchase_contract_effect = yes

		# Partner less happy but accepts
		reverse_add_opinion = {
			target = scope:merchant_partner
			modifier = annoyed_opinion
			opinion = -10
		}

		add_prestige = 100
		add_gold = 100

		# Stress if generous
		if = {
			limit = { has_trait = generous }
			add_stress = 15
		}

		ai_chance = {
			base = 30

			modifier = {
				add = 40
				has_trait = ambitious
			}

			modifier = {
				add = 25
				has_trait = arrogant
			}

			modifier = {
				add = -40
				has_trait = humble
			}
		}
	}

	# Option C: Decline partnership
	option = {
		name = mt_contract_events.0002.c

		# Missed opportunity
		add_prestige = -25

		reverse_add_opinion = {
			target = scope:merchant_partner
			modifier = disappointed_opinion
			opinion = -15
		}

		ai_chance = {
			base = 10

			modifier = {
				add = 30
				has_character_modifier = mt_notorious_contract_breaker
			}
		}
	}
}

# 0004: Foreign power wants trade agreement
mt_contract_events.0004 = {
	type = character_event
	title = mt_contract_events.0004.t
	desc = mt_contract_events.0004.desc
	theme = diplomacy

	left_portrait = {
		character = root
		animation = personality_bold
	}

	right_portrait = {
		character = scope:foreign_ruler
		animation = personality_honorable
	}

	trigger = {
		is_any_merchant_trigger = yes
		NOT = { has_variable = mt_foreign_contract }
	}

	immediate = {
		# Find foreign ruler
		random_neighboring_top_liege_realm_owner = {
			limit = {
				NOT = { this = root.liege }
			}
			save_scope_as = foreign_ruler
		}
	}

	# Option A: Accept - lucrative but politically risky
	option = {
		name = mt_contract_events.0004.a

		# Create contract
		set_variable = {
			name = mt_foreign_contract
			value = yes
		}

		set_variable = {
			name = contract_partner
			value = scope:foreign_ruler
		}

		set_variable = {
			name = contract_duration
			value = 5
		}

		set_variable = {
			name = contract_price
			value = 120
		}

		create_supply_contract_effect = yes

		# Large payment
		add_gold = 500

		# But your liege may not like this
		if = {
			limit = {
				exists = liege
				liege = {
					is_at_war_with = scope:foreign_ruler
				}
			}

			reverse_add_opinion = {
				target = liege
				modifier = betrayed_opinion
				opinion = -40
			}

			add_character_modifier = {
				modifier = mt_trading_with_enemy
				years = 5
			}
		}
		else_if = {
			limit = { exists = liege }

			reverse_add_opinion = {
				target = liege
				modifier = suspicious_opinion
				opinion = -15
			}
		}

		# Good relationship with foreign power
		add_opinion = {
			target = scope:foreign_ruler
			modifier = mt_business_partner_opinion
			opinion = 40
		}

		add_prestige = 150

		ai_chance = {
			base = 40

			modifier = {
				add = 40
				has_trait = greedy
			}

			modifier = {
				add = -100
				exists = liege
				liege = {
					is_at_war_with = scope:foreign_ruler
				}
			}

			modifier = {
				add = 30
				has_trait = ambitious
			}
		}
	}

	# Option B: Decline - maintain political neutrality
	option = {
		name = mt_contract_events.0004.b

		# Good standing with liege
		if = {
			limit = { exists = liege }
			reverse_add_opinion = {
				target = liege
				modifier = loyal_opinion
				opinion = 20
			}

			add_prestige = 75
		}

		# But missed huge opportunity
		add_gold = -50

		# Foreign ruler displeased
		reverse_add_opinion = {
			target = scope:foreign_ruler
			modifier = disappointed_opinion
			opinion = -20
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 50
				has_trait = just
			}

			modifier = {
				add = 40
				has_trait = loyal
			}

			modifier = {
				add = -40
				has_trait = greedy
			}
		}
	}

	# Option C: Secret deal (risky!)
	option = {
		name = mt_contract_events.0004.c

		trigger = {
			intrigue >= 12
		}

		# Secret contract
		set_variable = {
			name = mt_foreign_contract_secret
			value = yes
		}

		set_variable = {
			name = contract_partner
			value = scope:foreign_ruler
		}

		create_supply_contract_effect = yes

		# Large payment
		add_gold = 400

		# Secret modifier
		add_character_modifier = {
			modifier = mt_secret_foreign_trade
			years = 5
		}

		# Risk of discovery
		add_character_modifier = {
			modifier = mt_hidden_dealings
			years = 5
		}

		# Stress if honest
		if = {
			limit = { has_trait = honest }
			add_stress = 40
		}

		ai_chance = {
			base = 20

			modifier = {
				add = 40
				intrigue >= 16
			}

			modifier = {
				add = -100
				has_trait = honest
			}

			modifier = {
				add = 30
				has_trait = deceitful
			}
		}
	}
}

# 0005: Contract bidding war - multiple offers

	type = character_event
	title = mt_contract_events.0005.t
	desc = mt_contract_events.0005.desc
	theme = stewardship

		create_character = {

			name = contract_partner

		}

		set_variable = {
			name = contract_duration
			value = 3
		}

		set_variable = {
			name = contract_price
			value = 150
		}

		create_supply_contract_effect = yes

		# Large upfront bonus
		add_gold = 300

		# Other bidder angry
		scope:bidder_two = {
			add_opinion = {

				opinion = -25
			}
		}

		ai_chance = {
			base = 60

			modifier = {
				add = 40
				has_trait = greedy
			}

			modifier = {
				add = -20
				has_trait = generous
			}
		}
	}

	# Option B: Choose more reliable partner (lower bid)
	option = {
		name = mt_contract_events.0005.b

		# Safer terms
		set_variable = {
			name = contract_partner
			value = scope:bidder_two
		}

		set_variable = {
			name = contract_duration
			value = 5
		}

		set_variable = {
			name = contract_price
			value = 90
		}

		create_supply_contract_effect = yes

		# Moderate payment
		add_gold = 200

		# Build trust
		add_opinion = {
			target = scope:bidder_two
			modifier = mt_trusted_partner_opinion
			opinion = 30
		}

		# Reputation boost for wise choice
		if = {
			limit = { has_variable = merchant_reputation }
			change_variable = {
				name = merchant_reputation
				add = 15
			}
		}

		ai_chance = {
			base = 30

			modifier = {
				add = 40
				has_trait = cautious
			}

			modifier = {
				add = 30
				has_trait = just
			}
		}
	}

	# Option C: Play them against each other (risky!)
	option = {
		name = mt_contract_events.0005.c

		trigger = {
			diplomacy >= 14
		}

		# Try to get even better terms
		random = {
			chance = 60

			# Success - amazing deal
			set_variable = {
				name = contract_partner
				value = scope:bidder_one
			}

			set_variable = {
				name = contract_duration
				value = 3
			}

			set_variable = {
				name = contract_price
				value = 200
			}

			create_supply_contract_effect = yes

			add_gold = 500
			add_prestige = 150

				add_opinion = {
					target = root
					modifier = annoyed_opinion
					opinion = -15
				}
			}

			scope:bidder_two = {
				add_opinion = {
					target = root
					modifier = angry_opinion
					opinion = -30
				}
			}
		}

		# Failure - both withdraw
		random = {
			chance = 40

			# Lost both deals
			add_prestige = -100

			# Reputation damage
			if = {
				limit = { has_variable = merchant_reputation }
				change_variable = {
					name = merchant_reputation
					subtract = 20
				}
			}

			# Both bidders angry
			scope:bidder_one = {
				add_opinion = {
					target = root
					modifier = angry_opinion
					opinion = -40
				}
			}

			scope:bidder_two = {
				add_opinion = {
					target = root
					modifier = angry_opinion
					opinion = -40
				}
			}

			add_character_modifier = {
				modifier = mt_overplayed_hand
				years = 3
			}
		}

		# Stress if honest
		if = {
			limit = { has_trait = honest }
			add_stress = 20
		}

		ai_chance = {
			base = 10

			modifier = {
				add = 30
				has_trait = greedy
			}

			modifier = {
				add = 20
				diplomacy >= 16
			}

			modifier = {
				add = -50
				has_trait = honest
			}
		}
	}
}

	type = character_event
	title = mt_contract_events.0013.t
	desc = mt_contract_events.0013.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = personality_bold
	}

	right_portrait = {
		character = scope:delighted_partner
		animation = happiness
	}

	trigger = {
		has_variable = mt_supply_contract_active
		has_variable = mt_contract_months_fulfilled
		var:mt_contract_months_fulfilled >= 6
	}

	immediate = {
		if = {
			limit = { has_variable = mt_supply_contract_partner }
			var:mt_supply_contract_partner = {
				save_scope_as = delighted_partner
			}
		}
	}

	# Option A: Accept bonus and build on success
	option = {
		name = mt_contract_events.0013.a

		# Large bonus
		add_gold = 300
		add_prestige = 150

		# Reputation boost
		if = {
			limit = { has_variable = merchant_reputation }
			change_variable = {
				name = merchant_reputation
				add = 20
			}
		}

		# Excellent relationship
		add_opinion = {
			target = scope:delighted_partner
			modifier = mt_reliable_partner_opinion
			opinion = 30
		}

		reverse_add_opinion = {
			target = scope:delighted_partner
			modifier = grateful_opinion
			opinion = 40
		}

		# Achievement modifier
		add_character_modifier = {
			modifier = mt_exceeds_expectations
			years = 10
		}

		# Reduce stress
		add_stress = -25

		ai_chance = {
			base = 100
		}
	}
}

				add = 40

			}

			modifier = {
				add = -50
				has_trait = greedy
			}

			modifier = {
				add = 30
				opinion = {
					target = scope:renegotiating_partner
					value >= 50
				}
			}
		}
	}

		name = mt_contract_events.0014.b

			target = scope:renegotiating_partner
			modifier = respect_opinion
			opinion = 15
		}

		add_prestige = 75

		ai_chance = {
			base = 50

			modifier = {
				add = 30
				diplomacy >= 14
			}
		}
	}

	# Option C: Refuse - stick to original terms
	option = {
		name = mt_contract_events.0014.c

		# Keep current payment
		add_prestige = 100

		# But damage relationship
		reverse_add_opinion = {
			target = scope:renegotiating_partner
			modifier = angry_opinion
			opinion = -25
		}

		# Future renegotiation risk
		add_character_modifier = {
			modifier = mt_inflexible_contractor
			years = 5
		}

		ai_chance = {
			base = 10

			modifier = {
				add = 40
				has_trait = stubborn
			}

			modifier = {
				add = 30
				has_trait = greedy
			}

			modifier = {
				add = -40
				has_trait = generous
			}

# 0021: Can't fulfill contract - must breach
mt_contract_events.0021 = {
	type = character_event
	title = mt_contract_events.0021.t
	desc = mt_contract_events.0021.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = despair
	}

	right_portrait = {
		character = scope:wronged_partner
		animation = rage
	}

	trigger = {
		has_variable = mt_supply_contract_active
		has_character_modifier = mt_considering_contract_breach
	}

	immediate = {
		if = {
			limit = { has_variable = mt_supply_contract_partner }
			var:mt_supply_contract_partner = {
				save_scope_as = wronged_partner
			}
		}

		# Calculate contract value
		set_variable = {
			name = contract_value
			value = 1000
		}
	}

	# Option A: Breach and face the consequences
	option = {
		name = mt_contract_events.0021.a

		# Apply breach effect
		breach_contract_effect = yes

		# Extreme stress if honest/just
		if = {
			limit = {
				OR = {
					has_trait = honest
					has_trait = just
				}
			}
			add_stress = 60
		}
		else = {
			add_stress = 30
		}

		# Partner records the breach
		scope:wronged_partner = {
			set_variable = {
				name = mt_contract_breached_by
				value = root
			}
		}

		ai_chance = {
			base = 30

			modifier = {
				add = 60
				gold < 100
			}

			modifier = {
				add = -100
				has_trait = honest
			}

			modifier = {
				add = 50
				has_trait = callous
			}
		}
	}

	# Option B: Find substitute supplier (expensive)
	option = {
		name = mt_contract_events.0021.b

		trigger = {
			gold >= 500
			diplomacy >= 12
		}

		# Very expensive to arrange
		remove_short_term_gold = 500

		# But save the contract
		remove_character_modifier = mt_considering_contract_breach

		# Good relationship preserved
		reverse_add_opinion = {
			target = scope:wronged_partner
			modifier = relieved_opinion
			opinion = 15
		}

		# Reputation boost for resourcefulness
		if = {
			limit = { has_variable = merchant_reputation }
			change_variable = {
				name = merchant_reputation
				add = 10
			}
		}

		add_prestige = 100

		# Reduce stress from solving problem
		add_stress = -20

		ai_chance = {
			base = 50

			modifier = {
				add = 40
				gold >= 800
			}

			modifier = {
				add = 30
				diplomacy >= 14
			}

			modifier = {
				add = -60
				gold < 600
			}
		}
	}

	# Option C: Beg for extension
	option = {
		name = mt_contract_events.0021.c

		trigger = {
			diplomacy >= 10
		}

		# Check relationship
		if = {
			limit = {
				scope:wronged_partner = {
					opinion = {
						target = root
						value >= 40
					}
				}
			}

			# Extension granted
			add_character_modifier = {
				modifier = mt_contract_extension_granted
				months = 6
			}

			remove_character_modifier = mt_considering_contract_breach

			reverse_add_opinion = {
				target = scope:wronged_partner
				modifier = grateful_opinion
				opinion = 20
			}

			add_stress = -25

			custom_tooltip = mt_contract_events.0021.c.success
		}
		else = {
			# Extension denied - must breach
			breach_contract_effect = yes

			add_stress = 50
			add_prestige = -100

			custom_tooltip = mt_contract_events.0021.c.failure
		}

		# Stress from humiliation
		if = {
			limit = { has_trait = proud }
			add_stress = 30
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 50
				opinion = {
					target = scope:wronged_partner
					value >= 40
				}
			}

			modifier = {
				add = -60
				has_trait = proud
