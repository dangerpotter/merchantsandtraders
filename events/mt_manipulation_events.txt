# 0001: Successfully corner silk market - control prices
mt_manipulation.0001 = {
	type = character_event
	title = mt_manipulation.0001.t
	desc = mt_manipulation.0001.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		has_character_modifier = mt_market_monopoly_active

		base = 100

	# Maximize profits - set markup to 200%

		name = mt_manipulation.0001.a

		add_prestige = 100

				flag = under_market_investigation

		custom_tooltip = mt_manipulation.0001.b.tt

		name = mt_manipulation.0001.c

# 0002: Attempted corner fails - rivals unite against you
mt_manipulation.0002 = {
	type = character_event
	title = mt_manipulation.0002.t
	desc = mt_manipulation.0002.desc
	theme = intrigue

	left_portrait = {
		character = root
		animation = worry
	}

	right_portrait = {
		character = scope:rival_merchant
		animation = schadenfreude
	}

	trigger = {
		has_character_modifier = mt_market_monopoly_active
	}

	immediate = {
		# Find a rival merchant
		random_living_character = {
			limit = {
				is_any_merchant_trigger = yes
				NOT = { this = root }
			}
			save_scope_as = rival_merchant
		}
	}

	weight_multiplier = {
		base = 100
	}

	# Fight back - increase investment
	option = {
		name = mt_manipulation.0002.a

		remove_long_term_gold = 500

		# Risky - might succeed or fail catastrophically
		random_list = {
			50 = {
				# Success - maintain monopoly
				add_gold = 1000
				custom_tooltip = mt_manipulation.0002.a.success
			}
			50 = {
				# Failure - lose everything
				remove_character_modifier = mt_market_monopoly_active
				add_prestige = -200
				custom_tooltip = mt_manipulation.0002.a.failure
			}
		}

		ai_chance = {
			base = 40
			modifier = {
				add = 30
				has_trait = ambitious
			}
		}
	}

	# Negotiate with rivals - share the market
	option = {
		name = mt_manipulation.0002.b

		remove_short_term_gold = 300

		# Form cartel instead
		remove_character_modifier = mt_market_monopoly_active
		add_character_modifier = {
			modifier = mt_cartel_member
			years = 3
		}

		add_gold = 400

		scope:rival_merchant = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 30
			}
		}

		custom_tooltip = mt_manipulation.0002.b.tt

		ai_chance = {
			base = 40
			modifier = {
				add = 20
				diplomacy >= 12
			}
		}
	}

	# Surrender - cut your losses
	option = {
		name = mt_manipulation.0002.c

		remove_character_modifier = mt_market_monopoly_active
		add_prestige = -100

		# Avoid total ruin
		add_gold = 200

		if = {
			limit = { has_trait = ambitious }
			add_stress = 75
		}

		custom_tooltip = mt_manipulation.0002.c.tt

		ai_chance = {
			base = 20
		}
	}
}

	immediate = {

		random_living_character = {
			limit = {
				is_any_merchant_trigger = yes
				NOT = { this = root }
			}
			save_scope_as = new_competitor
		}
	}

	weight_multiplier = {
		base = 100
	}

	# Try to buy them out
	option = {
		name = mt_manipulation.0004.a

		remove_long_term_gold = 600

		random_list = {
			60 = {
				# Success
				add_gold = 300
				custom_tooltip = mt_manipulation.0004.a.success

				scope:new_competitor = {
					add_opinion = {
						target = root
						modifier = grateful_opinion
						opinion = 40
					}
				}
			}
			40 = {
				# They refuse
				custom_tooltip = mt_manipulation.0004.a.failure
				remove_character_modifier = mt_market_monopoly_active
				add_prestige = -100
			}
		}

		ai_chance = {
			base = 50
			modifier = {
				factor = 0
				gold < 700
			}
		}
	}

	# Compete aggressively - price war
	option = {
		name = mt_manipulation.0004.b

		# Initiate dumping campaign
		dump_goods_effect = yes

		scope:new_competitor = {
			add_opinion = {
				target = root
				modifier = rival_opinion
				opinion = -50
			}
		}

		custom_tooltip = mt_manipulation.0004.b.tt

		ai_chance = {
			base = 30
			modifier = {
				add = 30
				has_trait = vengeful
			}
		}
	}

	# Accept the competition
	option = {
		name = mt_manipulation.0004.c

		remove_character_modifier = mt_market_monopoly_active

		# Still profitable, just not monopoly profits
		add_gold = 300
		add_piety = 30

		scope:new_competitor = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 20
			}
		}

		if = {
			limit = { has_trait = greedy }
			add_stress = 50
		}

		ai_chance = {
			base = 20
			modifier = {
				add = 30
				has_trait = humble
			}
		}
	}
}

		animation = disapproval
	}

		has_character_flag = under_market_investigation

	}

	# Comply with government order
	option = {
		name = mt_manipulation.0005.a

		remove_long_term_gold = 300

		# Avoid worse punishment
		remove_character_flag = under_market_investigation

		reverse_add_opinion = {
			target = liege
			modifier = trust_opinion
			opinion = 10
		}

		custom_tooltip = mt_manipulation.0005.a.tt

		ai_chance = {
			base = 60
		}
	}

	# Bribe officials to look the other way
	option = {
		name = mt_manipulation.0005.b

		remove_short_term_gold = 400

		random_list = {
			50 = {
				# Bribe works
				remove_character_flag = under_market_investigation
				custom_tooltip = mt_manipulation.0005.b.success

				# Maintain monopoly
				add_gold = 500
			}
			50 = {
				# Bribe exposed - scandal
				trigger_event = mt_manipulation.0070
				custom_tooltip = mt_manipulation.0005.b.failure
			}
		}

		ai_chance = {
			base = 20
			modifier = {
				add = 30
				intrigue >= 12
			}
			modifier = {
				add = -50
				has_trait = just
			}
		}
	}

	# Defy the government
	option = {
		name = mt_manipulation.0005.c

		# Severe consequences
		reverse_add_opinion = {
			target = liege
			modifier = angry_opinion
			opinion = -40
		}

		add_prestige = -200
		remove_long_term_gold = 500

		# Possible imprisonment
		random = {
			chance = 40
			imprison = {
				target = root
				type = dungeon
			}
		}

		if = {
			limit = { has_trait = ambitious }
			add_stress = -50
		}

		custom_tooltip = mt_manipulation.0005.c.tt

		ai_chance = {
			base = 10
			modifier = {
				add = 20
				has_trait = arrogant
			}
		}
	}
}

# 0006: Other merchants boycott you
mt_manipulation.0006 = {
	type = character_event
	title = mt_manipulation.0006.t
	desc = mt_manipulation.0006.desc
	theme = diplomacy

	left_portrait = {
		character = root
		animation = worry
	}

	trigger = {
		has_character_modifier = mt_market_monopoly_active
		any_living_character = {
			is_any_merchant_trigger = yes
			NOT = { this = root }
		}
	}

	weight_multiplier = {
		base = 100
	}

	immediate = {
		every_living_character = {
			limit = {
				is_any_merchant_trigger = yes
				NOT = { this = root }
			}
			add_opinion = {
				target = root
				modifier = monopolist_opinion
				opinion = -30
			}
		}
	}

	# End monopoly to restore relations
	option = {
		name = mt_manipulation.0006.a

		remove_character_modifier = mt_market_monopoly_active

		every_living_character = {
			limit = {
				is_any_merchant_trigger = yes
				has_opinion_modifier = {
					target = root
					modifier = monopolist_opinion
				}
			}
			remove_opinion = {
				target = root
				modifier = monopolist_opinion
			}
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 20
			}
		}

		add_piety = 50

		custom_tooltip = mt_manipulation.0006.a.tt

		ai_chance = {
			base = 40
			modifier = {
				add = 30
				diplomacy >= 12
			}
		}
	}

	# Ignore the boycott
	option = {
		name = mt_manipulation.0006.b

		# Continue profiting but isolated
		add_gold = 400
		add_prestige = -50

		if = {
			limit = { has_trait = gregarious }
			add_stress = 50
		}

		custom_tooltip = mt_manipulation.0006.b.tt

		ai_chance = {
			base = 60
			modifier = {
				add = 30
				has_trait = greedy
			}
		}
	}
}
