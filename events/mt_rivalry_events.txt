# 0001: New merchant appears in your region - introduction event
mt_rivalry.0001 = {
	type = character_event
	title = mt_rivalry.0001.t
	desc = mt_rivalry.0001.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:new_rival
		animation = personality_bold
	}

	trigger = {
		OR = {
			has_trait = trade_post_owner
			has_trait = trading_company_head
			has_trait = merchant_prince
		}
		any_courtier = {
			OR = {
				has_trait = trade_post_owner
				has_trait = trading_company_head
			}
			NOT = { this = root }
			NOT = { has_relation_rival = root }
		}
	}

	weight_multiplier = {
		base = 1

		modifier = {
			add = 2
			has_character_modifier = mt_reputation_established_merchant
		}
	}

	immediate = {
		random_courtier = {
			limit = {
				OR = {
					has_trait = trade_post_owner
					has_trait = trading_company_head
				}
				NOT = { this = root }
				NOT = { has_relation_rival = root }
			}
			save_scope_as = new_rival
		}
	}

	# Greet them professionally - cautious approach
	option = {
		name = mt_rivalry.0001.a

		add_prestige = 10

		scope:new_rival = {
			add_opinion = {
				target = root
				modifier = trade_partner_opinion
				opinion = 10
			}
		}

		custom_tooltip = mt_rivalry.0001.a.tt

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				has_trait = diplomat
			}

			modifier = {
				add = 20
				has_trait = gregarious
			}
		}
	}

	# Keep your distance - watch them carefully
	option = {
		name = mt_rivalry.0001.b

		add_intrigue = 1

		add_character_modifier = {
			modifier = mt_watching_rival
			months = 12
		}

		custom_tooltip = mt_rivalry.0001.b.tt

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				has_trait = paranoid
			}

			modifier = {
				add = 20
				has_trait = diligent
			}
		}
	}

	# Immediately assert dominance - aggressive
	option = {
		name = mt_rivalry.0001.c

		scope:new_rival = {
			add_opinion = {
				target = root
				modifier = arrogant_opinion
				opinion = -20
			}
		}

		add_prestige = 25

		stress_impact = {
			humble = minor_stress_impact_gain
			compassionate = minor_stress_impact_gain
		}

		custom_tooltip = mt_rivalry.0001.c.tt

		ai_chance = {
			base = 20

			modifier = {
				add = 40
				has_trait = arrogant
			}

			modifier = {
				add = 30
				has_trait = ambitious
			}
		}
	}
}

# 0002: Existing merchant encroaching on your routes - warning
mt_rivalry.0002 = {
	type = character_event
	title = mt_rivalry.0002.t
	desc = mt_rivalry.0002.desc
	theme = stewardship

	left_portrait = {
		character = root
		animation = anger
	}

	right_portrait = {
		character = scope:encroaching_merchant
		animation = dismissal
	}

	trigger = {
		OR = {
			has_trait = trade_post_owner
			has_trait = trading_company_head
			has_trait = merchant_prince
		}
		any_courtier = {
			OR = {
				has_trait = trade_post_owner
				has_trait = trading_company_head
			}
			NOT = { this = root }
		}
	}

	weight_multiplier = {
		base = 1

		modifier = {
			add = 2
			any_courtier = {
				has_relation_rival = root
			}
		}
	}

	immediate = {
		random_courtier = {
			limit = {
				OR = {
					has_trait = trade_post_owner
					has_trait = trading_company_head
				}
				NOT = { this = root }
			}
			save_scope_as = encroaching_merchant
		}
	}

	# Confront them directly - demand they back off
	option = {
		name = mt_rivalry.0002.a

		duel = {
			skill = diplomacy
			target = scope:encroaching_merchant

			50 = { # You win
				desc = mt_rivalry.0002.a.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
				}

				scope:encroaching_merchant = {
					add_opinion = {
						target = root
						modifier = intimidated_opinion
						opinion = -15
					}
				}

				add_prestige = 50
				add_gold = 50
			}

			50 = { # You lose
				desc = mt_rivalry.0002.a.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3
				}

				scope:encroaching_merchant = {
					add_opinion = {
						target = root
						modifier = dismissed_opinion
						opinion = -25
					}
				}

				add_prestige = -25
				remove_short_term_gold = 50
			}
		}

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				diplomacy >= 14
			}

			modifier = {
				add = 20
				has_trait = arrogant
			}
		}
	}

	# Negotiate territory - divide the routes
	option = {
		name = mt_rivalry.0002.b

		scope:encroaching_merchant = {
			add_opinion = {
				target = root
				modifier = trade_partner_opinion
				opinion = 15
			}
		}

		add_gold = 100

		custom_tooltip = mt_rivalry.0002.b.tt

		ai_chance = {
			base = 40

			modifier = {
				add = 30
				has_trait = just
			}

			modifier = {
				add = 20
				has_trait = content
			}
		}
	}

	# Prepare for competition - strengthen your position
	option = {
		name = mt_rivalry.0002.c

		add_character_modifier = {
			modifier = mt_strengthening_position
			months = 18
		}

		remove_short_term_gold = 75

		custom_tooltip = mt_rivalry.0002.c.tt

		ai_chance = {
			base = 20

			modifier = {
				add = 30
				has_trait = patient
			}

			modifier = {
				add = 20
				stewardship >= 14
			}
		}
	}
