# ==============================================================================
# GLOBAL ECONOMY INITIALIZATION - NEW GAMES ONLY
# ==============================================================================
# Fires once per new game to establish the entire merchant economy
# Sets up global variables for market simulation and price tracking
# ==============================================================================
on_game_start_after_lobby = {
	on_actions = {
		mt_initialize_global_economy
	}
}

mt_initialize_global_economy = {
	trigger = {
		# Only run once per save file
		NOT = { has_global_variable = mt_economy_initialized }
	}

	effect = {
		# ======================================================================
		# MARK ECONOMY AS INITIALIZED
		# ======================================================================
		set_global_variable = {
			name = mt_economy_initialized
			value = yes
		}

		# ======================================================================
		# INITIALIZE GLOBAL MARKET PRICES
		# ======================================================================
		# Set baseline prices for all trade goods (scale: 1-100)
		# These fluctuate based on supply/demand events
		# ======================================================================

		# LUXURY GOODS - High base prices (60-90)
		set_global_variable = { name = mt_price_silk value = 80 }
		set_global_variable = { name = mt_price_spices value = 75 }
		set_global_variable = { name = mt_price_jade value = 85 }
		set_global_variable = { name = mt_price_pearls value = 90 }
		set_global_variable = { name = mt_price_gemstones value = 88 }
		set_global_variable = { name = mt_price_dyes value = 70 }
		set_global_variable = { name = mt_price_incense value = 65 }
		set_global_variable = { name = mt_price_perfume value = 72 }
		set_global_variable = { name = mt_price_ivory value = 78 }
		set_global_variable = { name = mt_price_amber value = 68 }
		set_global_variable = { name = mt_price_fine_wine value = 62 }
		set_global_variable = { name = mt_price_glassware value = 60 }

		# COMMON LUXURY - Medium-high prices (40-60)
		set_global_variable = { name = mt_price_wine value = 50 }
		set_global_variable = { name = mt_price_fur value = 55 }
		set_global_variable = { name = mt_price_wool value = 45 }
		set_global_variable = { name = mt_price_leather value = 42 }
		set_global_variable = { name = mt_price_salt value = 48 }
		set_global_variable = { name = mt_price_honey value = 46 }
		set_global_variable = { name = mt_price_wax value = 44 }

		# COMMON GOODS - Low-medium prices (20-40)
		set_global_variable = { name = mt_price_grain value = 25 }
		set_global_variable = { name = mt_price_timber value = 30 }
		set_global_variable = { name = mt_price_iron value = 38 }
		set_global_variable = { name = mt_price_copper value = 35 }
		set_global_variable = { name = mt_price_tin value = 32 }
		set_global_variable = { name = mt_price_fish value = 28 }
		set_global_variable = { name = mt_price_pottery value = 22 }
		set_global_variable = { name = mt_price_textiles value = 36 }

		# BULK GOODS - Very low prices (10-20)
		set_global_variable = { name = mt_price_stone value = 18 }
		set_global_variable = { name = mt_price_clay value = 12 }
		set_global_variable = { name = mt_price_rope value = 15 }
		set_global_variable = { name = mt_price_tar value = 14 }

		# ======================================================================
		# INITIALIZE MARKET TREND VARIABLES
		# ======================================================================
		# These track whether markets are rising or falling globally
		# Values: -2 (crash), -1 (decline), 0 (stable), 1 (growth), 2 (boom)
		# ======================================================================
		set_global_variable = { name = mt_market_trend_luxury value = 0 }
		set_global_variable = { name = mt_market_trend_common value = 0 }
		set_global_variable = { name = mt_market_trend_bulk value = 0 }

		# ======================================================================
		# INITIALIZE ECONOMIC CYCLE TRACKING
		# ======================================================================
		# Track current year in economic cycle (7-year cycles typical)
		set_global_variable = { name = mt_economic_cycle_year value = 1 }

		# Current economic state
		# Values: 0 (depression), 1 (recession), 2 (stable), 3 (growth), 4 (boom)
		set_global_variable = { name = mt_economic_state value = 2 }

		# ======================================================================
		# INITIALIZE SUPPLY/DEMAND TRACKING
		# ======================================================================
		# Global supply levels for trade good categories (scale 0-100)
		# Default to healthy supply (60)
		# ======================================================================
		set_global_variable = { name = mt_supply_luxury value = 60 }
		set_global_variable = { name = mt_supply_common value = 60 }
		set_global_variable = { name = mt_supply_bulk value = 60 }

		# Global demand levels (scale 0-100, default 50)
		set_global_variable = { name = mt_demand_luxury value = 50 }
		set_global_variable = { name = mt_demand_common value = 50 }
		set_global_variable = { name = mt_demand_bulk value = 50 }

		# ======================================================================
		# INITIALIZE MERCHANT ECONOMY PARAMETERS
		# ======================================================================
		# Track total number of active merchants globally
		set_global_variable = { name = mt_total_merchants value = 0 }

		# Track total active trade routes
		set_global_variable = { name = mt_total_trade_routes value = 0 }

		# Global merchant reputation average (affects all merchants)
		set_global_variable = { name = mt_global_merchant_reputation value = 50 }

		# ======================================================================
		# INITIALIZE RANDOM EVENT WEIGHTS
		# ======================================================================
		# These control frequency of random economic events
		# Values 0-100 (higher = more frequent)
		# ======================================================================
		set_global_variable = { name = mt_event_weight_prosperity value = 30 }
		set_global_variable = { name = mt_event_weight_disaster value = 20 }
		set_global_variable = { name = mt_event_weight_shortage value = 25 }
		set_global_variable = { name = mt_event_weight_surplus value = 25 }

		# ======================================================================
		# INITIALIZE REGIONAL MODIFIERS (Can be expanded per-region later)
		# ======================================================================
		# Track which regions have special economic conditions
		# These are flags that can be set/unset as events fire
		# ======================================================================
		# Examples (currently unused, for future expansion):
		# set_global_variable = { name = mt_mediterranean_boom value = no }
		# set_global_variable = { name = mt_silk_road_disrupted value = no }
		# set_global_variable = { name = mt_baltic_trade_flourishing value = no }

		# ======================================================================
		# LOG INITIALIZATION
		# ======================================================================
		debug_log = "MT: Global merchant economy initialized successfully"
		debug_log = "MT: All market prices set to baseline values"
		debug_log = "MT: Economic simulation ready"
	}
}

# ==============================================================================
# MONTHLY PULSE - GLOBAL MARKET PRICE UPDATES
# ==============================================================================
# Fires every month to simulate market fluctuations
# Adjusts prices based on supply/demand and random events
# ==============================================================================
on_monthly_pulse = {
	on_actions = {
		mt_monthly_market_update
	}
}

mt_monthly_market_update = {
	trigger = {
		# Only if economy is initialized
		has_global_variable = mt_economy_initialized
	}

	effect = {
		# ======================================================================
		# SMALL RANDOM PRICE FLUCTUATIONS
		# ======================================================================
		# Each good has 20% chance to fluctuate slightly each month
		# This simulates natural market volatility
		# ======================================================================

		# Random luxury good price changes (+/- 1-3 points)
		random_list = {
			10 = { change_variable = { name = global_var:mt_price_silk add = 2 } }
			10 = { change_variable = { name = global_var:mt_price_silk add = -2 } }
			80 = { } # No change
		}

		random_list = {
			10 = { change_variable = { name = global_var:mt_price_spices add = 2 } }
			10 = { change_variable = { name = global_var:mt_price_spices add = -2 } }
			80 = { }
		}

		random_list = {
			10 = { change_variable = { name = global_var:mt_price_jade add = 2 } }
			10 = { change_variable = { name = global_var:mt_price_jade add = -2 } }
			80 = { }
		}

		# ======================================================================
		# APPLY MARKET TREND EFFECTS
		# ======================================================================
		# If a market trend is active, gradually shift prices in that direction
		# ======================================================================
		if = {
			limit = { global_var:mt_market_trend_luxury = 1 }
			# Luxury goods trending up
			change_variable = { name = global_var:mt_price_silk add = 1 }
			change_variable = { name = global_var:mt_price_spices add = 1 }
			change_variable = { name = global_var:mt_price_jade add = 1 }
			change_variable = { name = global_var:mt_price_pearls add = 1 }
		}
		else_if = {
			limit = { global_var:mt_market_trend_luxury = -1 }
			# Luxury goods trending down
			change_variable = { name = global_var:mt_price_silk add = -1 }
			change_variable = { name = global_var:mt_price_spices add = -1 }
			change_variable = { name = global_var:mt_price_jade add = -1 }
			change_variable = { name = global_var:mt_price_pearls add = -1 }
		}

		# ======================================================================
		# CLAMP PRICES TO REASONABLE RANGES
		# ======================================================================
		# Prevent prices from going too extreme
		# Luxury: 40-120, Common: 15-80, Bulk: 5-40
		# ======================================================================
		if = {
			limit = { global_var:mt_price_silk > 120 }
			set_global_variable = { name = mt_price_silk value = 120 }
		}
		if = {
			limit = { global_var:mt_price_silk < 40 }
			set_global_variable = { name = mt_price_silk value = 40 }
		}

		# (Additional clamping for other goods would go here in full implementation)

		# ======================================================================
		# TRIGGER RANDOM ECONOMIC EVENTS
		# ======================================================================
		# Small chance each month for major economic event
		# ======================================================================
		random_list = {
			# Market boom (2% chance)
			2 = {
				modifier = {
					add = 5
					global_var:mt_economic_state >= 3
				}
				# Trigger economic boom event
				trigger_event = {
					id = mt_economic_events.0100
					days = { 5 15 }
				}
			}

			# Market crash (2% chance)
			2 = {
				modifier = {
					add = 5
					global_var:mt_economic_state <= 1
				}
				# Trigger economic crash event
				trigger_event = {
					id = mt_economic_events.0200
					days = { 5 15 }
				}
			}

			# No major event (96%)
			96 = { }
		}
	}
}

# ==============================================================================
# YEARLY PULSE - ECONOMIC CYCLE PROGRESSION
# ==============================================================================
# Fires once per year to advance the economic cycle
# Handles long-term economic trends and resets
# ==============================================================================
on_yearly_pulse = {
	on_actions = {
		mt_yearly_economic_cycle
	}
}

mt_yearly_economic_cycle = {
	trigger = {
		has_global_variable = mt_economy_initialized
	}

	effect = {
		# ======================================================================
		# ADVANCE ECONOMIC CYCLE
		# ======================================================================
		change_variable = { name = global_var:mt_economic_cycle_year add = 1 }

		# Reset cycle after 7 years
		if = {
			limit = { global_var:mt_economic_cycle_year > 7 }
			set_global_variable = { name = mt_economic_cycle_year value = 1 }

			# Chance to shift economic state at cycle reset
			random_list = {
				20 = {
					# Shift toward growth
					if = {
						limit = { global_var:mt_economic_state < 4 }
						change_variable = { name = global_var:mt_economic_state add = 1 }
					}
				}
				20 = {
					# Shift toward recession
					if = {
						limit = { global_var:mt_economic_state > 0 }
						change_variable = { name = global_var:mt_economic_state add = -1 }
					}
				}
				60 = {
					# Remain stable
				}
			}
		}

		# ======================================================================
		# GRADUAL TREND REVERSION
		# ======================================================================
		# Market trends gradually return to neutral over time
		# ======================================================================
		if = {
			limit = { global_var:mt_market_trend_luxury > 0 }
			random = {
				chance = 30
				change_variable = { name = global_var:mt_market_trend_luxury add = -1 }
			}
		}
		else_if = {
			limit = { global_var:mt_market_trend_luxury < 0 }
			random = {
				chance = 30
				change_variable = { name = global_var:mt_market_trend_luxury add = 1 }
			}
		}

		# ======================================================================
		# SUPPLY/DEMAND REBALANCING
		# ======================================================================
		# Gradually return supply and demand toward equilibrium (50)
		# ======================================================================
		if = {
			limit = { global_var:mt_supply_luxury > 55 }
			change_variable = { name = global_var:mt_supply_luxury add = -5 }
		}
		else_if = {
			limit = { global_var:mt_supply_luxury < 45 }
			change_variable = { name = global_var:mt_supply_luxury add = 5 }
		}

		if = {
			limit = { global_var:mt_demand_luxury > 55 }
			change_variable = { name = global_var:mt_demand_luxury add = -5 }
		}
		else_if = {
			limit = { global_var:mt_demand_luxury < 45 }
			change_variable = { name = global_var:mt_demand_luxury add = 5 }
		}

		# ======================================================================
		# LOG YEARLY STATUS
		# ======================================================================
		debug_log = "MT: Yearly economic update completed"
		debug_log_scopes = yes
	}
}

# ==============================================================================
# CLEAN STATE ON NEW GAME
# ==============================================================================
# Additional check to ensure clean slate for new games
# Clears any lingering variables from previous saves
# ==============================================================================
on_game_start = {
	on_actions = {
		mt_ensure_clean_state
	}
}

mt_ensure_clean_state = {
	trigger = {
		# Only on first load of a completely new game
		NOT = { has_global_variable = mt_economy_initialized }
		NOT = { has_global_variable = mt_game_initialized }
	}

	effect = {
		# Clear any potential orphaned variables
		# (This is a safety measure for development/testing)

		# Remove any temporary debugging flags
		if = {
			limit = { has_global_variable = mt_debug_mode }
			remove_global_variable = mt_debug_mode
		}

		# Log clean state confirmation
		debug_log = "MT: Clean state confirmed for new game"
	}
}
