####################################
# MERCHANT REPUTATION SCRIPT VALUES
####################################
# Fame-like reputation system for merchants
# Reputation affects prices, opportunities, and prestige

#######################
# BASE REPUTATION GAIN
#######################

# Base reputation from successful major trade
reputation_successful_trade = {
	value = 10

	# Bonus for exceptional profit
	if = {
		limit = { has_character_flag = exceptional_profit_trade }
		add = 5
	}

	# Bonus for first trade in new region
	if = {
		limit = { has_character_flag = pioneering_trade }
		add = 10
	}
}

# Reputation from charitable acts
reputation_from_charity = {
	value = 5

	# Large donations worth more
	if = {
		limit = { has_character_flag = major_charitable_donation }
		add = 10
	}

	# Bonus if charity to poor
	if = {
		limit = { has_character_flag = charity_to_poor }
		add = 5
	}
}

# Reputation from honest dealing
reputation_from_honesty = {
	value = 15

	# Extra bonus if temptation was high
	if = {
		limit = { has_character_flag = resisted_major_fraud_opportunity }
		add = 10
	}
}

# Reputation from helping rulers
reputation_from_royal_service = {
	value = 20

	# Bonus for critical assistance
	if = {
		limit = { has_character_flag = saved_realm_financially }
		add = 30
	}
}

# Reputation from discovering trade routes
reputation_from_discovery = {
	value = 25

	# Major discovery worth more
	if = {
		limit = { has_character_flag = discovered_silk_road }
		add = 25
	}
}

#######################
# BASE REPUTATION LOSS
#######################

# Reputation loss from failed trade
reputation_loss_failed_trade = {
	value = -5

	# Worse if total loss
	if = {
		limit = { has_character_flag = catastrophic_trade_failure }
		add = -10
	}
}

# Reputation loss from bankruptcy
reputation_loss_bankruptcy = {
	value = -50

	# Even worse if second bankruptcy
	if = {
		limit = { has_variable = bankruptcy_count }
		if = {
			limit = {
				check_variable = {
					which = bankruptcy_count
					value >= 2
				}
			}
			add = -50
		}
	}
}

# Reputation loss from fraud
reputation_loss_fraud = {
	value = -100

	# Catastrophic if caught multiple times
	if = {
		limit = { has_character_flag = repeat_fraudster }
		add = -100
	}
}

# Reputation loss from breaking contracts
reputation_loss_broken_contract = {
	value = -30

	# Worse if contract with ruler
	if = {
		limit = { has_character_flag = broke_royal_contract }
		add = -30
	}
}

# Reputation loss from defaulting on loans
reputation_loss_loan_default = {
	value = -40

	# Worse for larger loans
	if = {
		limit = { has_character_flag = major_loan_default }
		add = -35
	}
}

#######################
# REPUTATION MODIFIERS
#######################

# Yearly reputation decay
reputation_decay_yearly = {
	value = -1

	# No decay if actively trading
	if = {
		limit = { has_character_flag = actively_trading }
		add = 1
	}

	# Slower decay at higher tiers
	if = {
		limit = { has_variable = merchant_reputation }
		if = {
			limit = {
				check_variable = {
					which = merchant_reputation
					value >= 300
				}
			}
			add = 0.5
		}
	}
}

# Reputation bonus from wealth
reputation_from_wealth = {
	value = 0

	# +1 per 1000 gold
	if = {
		limit = { gold >= 1000 }
		add = 1
	}
	if = {
		limit = { gold >= 2000 }
		add = 1
	}
	if = {
		limit = { gold >= 3000 }
		add = 1
	}
	if = {
		limit = { gold >= 5000 }
		add = 2
	}
	if = {
		limit = { gold >= 7500 }
		add = 2
	}
	if = {
		limit = { gold >= 10000 }
		add = 3
	}
}

# Reputation bonus from active profitable routes
reputation_from_routes = {
	value = 0

	# +2 per profitable route (tracked via variables)
	if = {
		limit = { has_variable = profitable_routes }
		add = profitable_routes
		multiply = 2
	}
}

# Reputation bonus from merchant tier
reputation_from_tier = {
	value = 0

	# Wandering Merchant: 0
	if = {
		limit = { has_character_modifier = wandering_merchant }
		add = 0
	}

	# Local Trader: 10
	if = {
		limit = { has_character_modifier = local_trader_tier }
		add = 10
	}

	# Regional Merchant: 20
	if = {
		limit = { has_character_modifier = regional_merchant }
		add = 20
	}

	# Trade Factor: 30
	if = {
		limit = { has_character_modifier = trade_factor }
		add = 30
	}

	# Merchant Prince: 50
	if = {
		limit = { has_character_modifier = merchant_prince }
		add = 50
	}
}

#######################
# TOTAL REPUTATION
#######################

# Calculate total merchant reputation
total_merchant_reputation = {
	value = 0

	# Base reputation from variable
	if = {
		limit = { has_variable = merchant_reputation }
		add = merchant_reputation
	}

	# Ongoing bonuses from wealth
	add = reputation_from_wealth

	# Ongoing bonuses from routes
	add = reputation_from_routes

	# Ongoing bonuses from tier
	add = reputation_from_tier
}
