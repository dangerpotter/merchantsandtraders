############################################################################################
# MERCHANTS AND TRADERS - GOODS TRIGGERS
############################################################################################
#
# This file contains scripted triggers for checking trade goods production and status.
# Triggers return yes/no and are used in conditions throughout the mod.
#
# TRIGGER DESIGN PRINCIPLES:
# - Fast evaluation - triggers are called frequently, keep them efficient
# - Clear naming - trigger names describe what they check
# - Flexible parameters - use $PARAMETER$ syntax for reusability
# - Scope-safe - document what scope (county, character, etc.) is expected
#
# INTEGRATION POINTS:
# - Use in event conditions to check if county produces specific goods
# - Use in AI evaluation to target valuable counties
# - Use in building requirements to unlock specialized structures
# - Use in decision conditions to gate economic actions
#
############################################################################################

############################################################################################
# COUNTY PRODUCES GOODS TRIGGER
############################################################################################
# ROOT: County
# Checks if a county produces a specific trade good
#
# PARAMETERS:
#   GOOD - The trade good to check (e.g., silk, spices, grain)
#
# USAGE:
#   any_county = {
#       county_produces_goods_trigger = { GOOD = silk }
#   }
#
# RETURNS: yes if county has the specified trade good modifier
#
# INTEGRATION:
# [HOOK] Events can check for goods before triggering trade events
# [HOOK] AI can evaluate county value based on goods produced
# [HOOK] Buildings can require specific goods to be constructed
############################################################################################

county_produces_goods_trigger = {
	has_county_modifier = mt_trade_good_$GOOD$
}

############################################################################################
# HAS LUXURY GOOD TRIGGER
############################################################################################
# ROOT: County
# Checks if a county produces any luxury trade good
#
# LUXURY GOODS: silk, spices, jade, amber, pearls, ivory, purple_dye, incense, saffron
#
# USAGE:
#   any_county = {
#       has_luxury_good_trigger = yes
#   }
#
# RETURNS: yes if county produces at least one luxury good
#
# INTEGRATION:
# [HOOK] AI can prioritize conquering luxury-producing counties
# [HOOK] Events can trigger based on luxury good availability
# [HOOK] Trade routes can calculate higher profits for luxury goods
# [HOOK] Diplomatic actions can reference luxury goods as gifts
############################################################################################

has_luxury_good_trigger = {
	OR = {
		has_county_modifier = mt_trade_good_silk
		has_county_modifier = mt_trade_good_spices
		has_county_modifier = mt_trade_good_jade
		has_county_modifier = mt_trade_good_amber
		has_county_modifier = mt_trade_good_pearls
		has_county_modifier = mt_trade_good_ivory
		has_county_modifier = mt_trade_good_purple_dye
		has_county_modifier = mt_trade_good_incense
		has_county_modifier = mt_trade_good_saffron
	}
}

############################################################################################
# GOODS PRODUCTION LEVEL TRIGGER
############################################################################################
# ROOT: County
# Checks if a county's production level matches a specific tier
#
# PARAMETERS:
#   LEVEL - Production level to check: 0=poor, 1=modest, 2=rich, 3=abundant
#
# USAGE:
#   any_county = {
#       goods_production_level_trigger = { LEVEL = 3 }  # Check for abundant
#   }
#
# RETURNS: yes if county's mt_production_level variable equals LEVEL
#
# INTEGRATION:
# [HOOK] Events can trigger based on prosperity or decline
# [HOOK] Buildings may require certain production levels
# [HOOK] AI can evaluate investment opportunities
# [HOOK] Trade route efficiency depends on production levels
############################################################################################

goods_production_level_trigger = {
	has_variable = mt_production_level
	var:mt_production_level = $LEVEL$
}

############################################################################################
# PRODUCES MULTIPLE GOODS TRIGGER
############################################################################################
# ROOT: County
# Checks if county produces multiple trade goods (diversified economy)
#
# PARAMETERS:
#   COUNT - Minimum number of goods required (default 2)
#
# USAGE:
#   any_county = {
#       produces_multiple_goods_trigger = { COUNT = 3 }
#   }
#
# RETURNS: yes if county produces at least COUNT different goods
#
# NOTE: This is a framework trigger. Full implementation would count active modifiers.
# Current version checks for common multi-good combinations.
#
# INTEGRATION:
# [HOOK] Economic prosperity events favor diversified counties
# [HOOK] AI rates diversified counties as more valuable
# [HOOK] Trade hubs tend to have multiple goods
############################################################################################

produces_multiple_goods_trigger = {
	# Framework implementation: check for presence of multiple goods
	# In full implementation, this would count active trade good modifiers

	# Check for at least 2 goods (simplified check)
	OR = {
		# Luxury + Common combination
		AND = {
			has_luxury_good_trigger = yes
			has_common_good_trigger = yes
		}
		# Luxury + Regional combination
		AND = {
			has_luxury_good_trigger = yes
			has_regional_good_trigger = yes
		}
		# Common + Regional combination
		AND = {
			has_common_good_trigger = yes
			has_regional_good_trigger = yes
		}
		# Multiple commons (example: grain + cattle)
		AND = {
			has_county_modifier = mt_trade_good_grain
			OR = {
				has_county_modifier = mt_trade_good_cattle
				has_county_modifier = mt_trade_good_wine
				has_county_modifier = mt_trade_good_iron
			}
		}
	}
}

############################################################################################
# CAN PRODUCE LUXURY GOODS TRIGGER
############################################################################################
# ROOT: County
# Checks if county has potential to produce luxury goods (geography/development)
#
# USAGE:
#   any_county = {
#       can_produce_luxury_goods_trigger = yes
#   }
#
# RETURNS: yes if county is in a region known for luxury goods or has high development
#
# INTEGRATION:
# [HOOK] Investment decisions can check potential
# [HOOK] AI can plan long-term development
# [HOOK] Events about discovering new resources
############################################################################################

can_produce_luxury_goods_trigger = {
	OR = {
		# Already produces luxury goods
		has_luxury_good_trigger = yes

		# High development suggests capacity for luxury production
		development_level >= 20

		# Located in known luxury-producing regions
		geographical_region = world_india
		geographical_region = world_persia
		geographical_region = world_steppe_east

		# Coastal Mediterranean (purple dye, glass)
		AND = {
			is_coastal_county = yes
			OR = {
				geographical_region = world_europe_south
				geographical_region = world_middle_east
			}
		}

		# Baltic coast (amber)
		AND = {
			is_coastal_county = yes
			geographical_region = world_europe_north
		}
	}
}
