############################################################################################
# MERCHANTS AND TRADERS - GOODS PRODUCTION SCRIPTED EFFECTS
############################################################################################
#
# This file contains scripted effects for managing the trade goods supply chain system.
#
# SUPPLY CHAIN CONCEPT:
# The trade goods system models a complete economic cycle:
# 1. PRODUCTION - Counties generate goods based on terrain, culture, and development
# 2. QUANTITY - Production levels determine output volume (poor/modest/rich/abundant)
# 3. VALUE - Each good has base value modified by supply/demand (see modifiers file)
# 4. TRADE - Goods flow along trade routes to markets (future integration)
# 5. CONSUMPTION - Goods are consumed, generating income and effects
#
# MODULAR DESIGN:
# - Effects are self-contained and can be called independently
# - Use variables to pass state between effects
# - Hook points marked with [INTEGRATION] for future systems
# - All effects scope-safe (clearly documented who ROOT/THIS refers to)
#
# KEY VARIABLES USED:
# - mt_production_level: 0=poor, 1=modest, 2=rich, 3=abundant
# - mt_goods_count: Number of goods county produces (1-3)
# - mt_goods_value: Calculated value of production
#
############################################################################################

############################################################################################
# ASSIGN COUNTY TRADE GOODS
############################################################################################
# ROOT: County
# Assigns 1-3 trade goods to a county based on its terrain, culture, and geographic features
# This is the foundation of the production system - determining what each county makes
#
# USAGE:
#   any_county = {
#       assign_county_trade_goods_effect = yes
#   }
#
# INTEGRATION POINTS:
# [HOOK] Can be called on game start to initialize all counties
# [HOOK] Can be called when province changes hands or is developed
# [HOOK] Can be modified by buildings to add specialized goods
############################################################################################

	# Based on terrain type (hills, plains, forest, etc.)

	if = {
		limit = {
			OR = {
				terrain = farmlands
				terrain = plains
			}
		}
		add_county_modifier = {
			modifier = mt_trade_good_grain
			years = -1  # Permanent until changed
		}
	}

	else_if = {
		limit = { terrain = forest }
		add_county_modifier = {
			modifier = mt_trade_good_timber
			years = -1
		}
		# 50% chance for furs as secondary good
		random = {
			chance = 50
			add_county_modifier = {
				modifier = mt_trade_good_furs
				years = -1
			}
		}
	}

	else_if = {

			chance = 30

############################################################################################
# UPDATE GOODS PRODUCTION LEVEL
############################################################################################
# ROOT: County
# Changes the production level of trade goods in a county
# Multiplies base modifier values based on production tier
#
# PRODUCTION LEVELS:
# - POOR (0): 0.5x base value - war, disease, bad weather, mismanagement
# - MODEST (1): 1.0x base value - normal production (default)
# - RICH (2): 1.5x base value - good conditions, investment, skilled workers
# - ABUNDANT (3): 2.0x base value - ideal conditions, advanced infrastructure
#
# USAGE:
#   any_county = {
#       set_variable = { name = mt_production_level value = 2 }  # Set to RICH
#       update_goods_production_effect = yes
#   }
#
# INTEGRATION POINTS:
# [HOOK] Development level can influence production tier
# [HOOK] Buildings can boost production level
# [HOOK] Events (plague, prosperity, innovation) can change level
# [HOOK] Trade route efficiency can affect production incentives
############################################################################################

update_goods_production_effect = {
	# Store the production level (default to MODEST if not set)
	if = {
		limit = { NOT = { has_variable = mt_production_level } }
		set_variable = { name = mt_production_level value = 1 }  # MODEST default
	}

	# POOR PRODUCTION (0.5x multiplier)
	# Conditions: War, low development, no investment
	if = {
		limit = { var:mt_production_level = 0 }

		# Apply 50% penalty to all economic modifiers
		# [TECHNICAL] In full implementation, this would iterate through active goods
		# and adjust their values. For now, this is a framework placeholder.
		# [INTEGRATION] County modifier values can be scaled via scripted effects

		add_county_modifier = {
			modifier = mt_production_poor
			years = -1
		}
		remove_county_modifier = mt_production_modest
		remove_county_modifier = mt_production_rich
		remove_county_modifier = mt_production_abundant
	}

	# MODEST PRODUCTION (1.0x multiplier - baseline)
	# Conditions: Normal peacetime operation
	else_if = {
		limit = { var:mt_production_level = 1 }

		# Base production, no modifiers needed
		add_county_modifier = {
			modifier = mt_production_modest
			years = -1
		}
		remove_county_modifier = mt_production_poor
		remove_county_modifier = mt_production_rich
		remove_county_modifier = mt_production_abundant
	}

	# RICH PRODUCTION (1.5x multiplier)
	# Conditions: Peace, good stewardship, some investment
	else_if = {
		limit = { var:mt_production_level = 2 }

		# Apply 50% bonus to production value
		add_county_modifier = {
			modifier = mt_production_rich
			years = -1
		}
		remove_county_modifier = mt_production_poor
		remove_county_modifier = mt_production_modest
		remove_county_modifier = mt_production_abundant
	}

	# ABUNDANT PRODUCTION (2.0x multiplier)
	# Conditions: Long peace, high development, infrastructure, skilled management
	else_if = {
		limit = { var:mt_production_level = 3 }

		# Apply 100% bonus to production value
		add_county_modifier = {
			modifier = mt_production_abundant
			years = -1
		}
		remove_county_modifier = mt_production_poor
		remove_county_modifier = mt_production_modest
		remove_county_modifier = mt_production_rich
	}

	# [INTEGRATION] Hook for trade route system to read production level
	# [INTEGRATION] Hook for AI to evaluate county value
	# [INTEGRATION] Hook for events to reference prosperity
}

############################################################################################
# CALCULATE GOODS BASE VALUE
############################################################################################
# ROOT: County or Character
# Calculates the total base value of all trade goods produced/controlled
# Stores result in mt_goods_value variable for use by other systems
#
# VALUE CALCULATION:
# - Sums the monthly_income from all active trade good modifiers
# - Applies production level multiplier
# - Can be modified by supply/demand (future integration)
# - Returns base gold value before trade route bonuses
#
# USAGE:
#   any_county = {
#       calculate_goods_base_value_effect = yes
#       # Now var:mt_goods_value contains total value
#   }
#
# INTEGRATION POINTS:
# [HOOK] Trade route system uses this to calculate profit
# [HOOK] AI uses this to evaluate county wealth
# [HOOK] Taxation system can reference this value
# [HOOK] Supply/demand system modifies base value
############################################################################################

calculate_goods_base_value_effect = {
	# Initialize value counter
	set_variable = { name = mt_goods_value value = 0 }

	# Add value from each luxury good (if present)
	if = {
		limit = { has_county_modifier = mt_trade_good_silk }
		change_variable = { name = mt_goods_value add = 2.5 }
	}

		limit = { has_county_modifier = mt_trade_good_spices }
