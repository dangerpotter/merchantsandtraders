# Effect: Calculate and update merchant reputation
# Scope: Character
# Usage: <character> = { calculate_merchant_reputation_effect = yes }
calculate_merchant_reputation_effect = {
	# Save current reputation value
	# (In full implementation, this would read from character variable)

	# Remove all existing reputation tier modifiers
	if = {
		limit = { has_character_modifier = mt_reputation_unknown_peddler }
		remove_character_modifier = mt_reputation_unknown_peddler
	}
	if = {
		limit = { has_character_modifier = mt_reputation_local_trader }
		remove_character_modifier = mt_reputation_local_trader
	}
	if = {
		limit = { has_character_modifier = mt_reputation_known_merchant }
		remove_character_modifier = mt_reputation_known_merchant
	}
	if = {
		limit = { has_character_modifier = mt_reputation_respected_trader }
		remove_character_modifier = mt_reputation_respected_trader
	}
	if = {
		limit = { has_character_modifier = mt_reputation_prominent_merchant }
		remove_character_modifier = mt_reputation_prominent_merchant
	}
	if = {
		limit = { has_character_modifier = mt_reputation_renowned_trader }
		remove_character_modifier = mt_reputation_renowned_trader
	}
	if = {
		limit = { has_character_modifier = mt_reputation_merchant_magnate }
		remove_character_modifier = mt_reputation_merchant_magnate
	}
	if = {
		limit = { has_character_modifier = mt_reputation_trade_master }
		remove_character_modifier = mt_reputation_trade_master
	}
	if = {
		limit = { has_character_modifier = mt_reputation_legend_of_commerce }
		remove_character_modifier = mt_reputation_legend_of_commerce
	}
	if = {
		limit = { has_character_modifier = mt_reputation_immortal_legacy }
		remove_character_modifier = mt_reputation_immortal_legacy
	}

	# Apply appropriate tier modifier based on reputation
	# (This uses character variables that would be set elsewhere)

	# Tier 1: Unknown Peddler (0-9)
	if = {
		limit = {
			OR = {
				NOT = { exists = var:merchant_reputation }
				var:merchant_reputation < 10
			}
		}
		add_character_modifier = {
			modifier = mt_reputation_unknown_peddler
		}
	}

	# Tier 2: Local Trader (10-24)
	else_if = {
		limit = {
			var:merchant_reputation >= 10
			var:merchant_reputation < 25
		}
		add_character_modifier = {
			modifier = mt_reputation_local_trader
		}
	}

	# Tier 3: Known Merchant (25-49)
	else_if = {
		limit = {
			var:merchant_reputation >= 25
			var:merchant_reputation < 50
		}
		add_character_modifier = {
			modifier = mt_reputation_known_merchant
		}
	}

	# Tier 4: Respected Trader (50-99)
	else_if = {
		limit = {
			var:merchant_reputation >= 50
			var:merchant_reputation < 100
		}
		add_character_modifier = {
			modifier = mt_reputation_respected_trader
		}
	}

	# Tier 5: Prominent Merchant (100-199)
	else_if = {
		limit = {
			var:merchant_reputation >= 100
			var:merchant_reputation < 200
		}
		add_character_modifier = {
			modifier = mt_reputation_prominent_merchant
		}
	}

	# Tier 6: Renowned Trader (200-299)
	else_if = {
		limit = {
			var:merchant_reputation >= 200
			var:merchant_reputation < 300
		}
		add_character_modifier = {
			modifier = mt_reputation_renowned_trader
		}
	}

	# Tier 7: Merchant Magnate (300-449)
	else_if = {
		limit = {
			var:merchant_reputation >= 300
			var:merchant_reputation < 450
		}
		add_character_modifier = {
			modifier = mt_reputation_merchant_magnate
		}
	}

	# Tier 8: Trade Master (450-599)
	else_if = {
		limit = {
			var:merchant_reputation >= 450
			var:merchant_reputation < 600
		}
		add_character_modifier = {
			modifier = mt_reputation_trade_master
		}
	}

	# Tier 9: Legend of Commerce (600-799)
	else_if = {
		limit = {
			var:merchant_reputation >= 600
			var:merchant_reputation < 800
		}
		add_character_modifier = {
			modifier = mt_reputation_legend_of_commerce
		}
	}

	# Tier 10: Immortal Legacy (800+)
	else_if = {
		limit = {
			var:merchant_reputation >= 800
		}
		add_character_modifier = {
			modifier = mt_reputation_immortal_legacy
		}
	}
}

# INITIALIZATION EFFECTS
####################################

# Effect: Initialize reputation system for new merchant
# Scope: Character
# Usage: <character> = { initialize_merchant_reputation_effect = yes }
initialize_merchant_reputation_effect = {
	# Set starting reputation based on tier
	if = {
		limit = { has_trait = wandering_merchant }
		set_variable = {
			name = merchant_reputation
			value = 0
		}
	}
	else_if = {
		limit = { has_trait = trade_post_owner }
		set_variable = {
			name = merchant_reputation
			value = 25
		}
	}
	else_if = {
		limit = { has_trait = trading_company_head }
		set_variable = {
			name = merchant_reputation
			value = 100
		}
	}
	else_if = {
		limit = { has_trait = merchant_prince }
		set_variable = {
			name = merchant_reputation
			value = 300
		}
	}
	else = {
		set_variable = {
			name = merchant_reputation
			value = 0
		}
	}

	# Apply initial tier modifier
	calculate_merchant_reputation_effect = yes
}

# PERIODIC EFFECTS

# Effect: Monthly reputation decay (if inactive)
# Scope: Character
# Usage: <character> = { reputation_decay_effect = yes }
reputation_decay_effect = {
	# Lose 1 reputation per year (0.083 per month)
	# Only apply if merchant is inactive
	if = {
		limit = {
			NOT = { has_character_flag = merchant_active_this_year }
		}
		change_variable = {
			name = merchant_reputation
			add = -1
		}

		# Ensure doesn't go below 0
		if = {
			limit = { var:merchant_reputation < 0 }
			set_variable = {
				name = merchant_reputation
				value = 0
			}
		}

		calculate_merchant_reputation_effect = yes
	}
}
