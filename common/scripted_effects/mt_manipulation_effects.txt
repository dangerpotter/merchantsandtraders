# MARKET CORNERING
####################################

# Effect: Attempt to monopolize a specific good
# Scope: Character
# Usage: <character> = { corner_market_effect = yes }
# Requires: 80% of regional supply, massive gold investment (500-2000)
# Effect: Control prices, set markup up to 200%
# Risk: Regulatory backlash, rival merchants unite
corner_market_effect = {
	# Remove existing gold based on market size
	remove_long_term_gold = 1000

	# Apply monopoly control modifier
	add_character_modifier = {
		modifier = mt_market_monopoly_active
		years = 5
	}

	# Add massive wealth from price control
	add_gold = 2000

	# Risk: Attract attention from authorities
	random = {
		chance = 40
		# Trigger regulatory investigation
		trigger_event = {
			id = mt_manipulation.0001
			days = { 180 360 }
		}
	}

	# Risk: Rival merchants unite against you
	random = {
		chance = 30
		trigger_event = {
			id = mt_manipulation.0002
			days = { 90 180 }
		}
	}

	# Reputation hit for unethical practices
	add_character_modifier = {
		modifier = mt_known_market_manipulator
		years = 10
	}

	# Market stress - maintain or lose control
	trigger_event = {
		id = mt_manipulation.0003
		days = { 365 730 }
	}
}

####################################
# RUMOR SPREADING
####################################

# Effect: Market manipulation via misinformation
# Scope: Character
# Usage: <character> = { spread_rumors_effect = yes }
# Success based on reputation and stewardship/intrigue
spread_rumors_effect = {
	# Small cost for spreading disinformation
	remove_short_term_gold = 50

	# Success check based on intrigue and reputation
	if = {
		limit = {
			intrigue >= 14
			OR = {
				has_character_modifier = mt_reputation_renowned_trader
				has_character_modifier = mt_reputation_merchant_magnate
			}
		}
		# High success - rumors believed
		trigger_event = mt_manipulation.0031

		# Add temporary price manipulation
		add_character_modifier = {
			modifier = mt_rumor_manipulation_successful
			months = 6
		}

		add_gold = 400
	}
	else_if = {
		limit = {
			intrigue >= 10
		}
		# Moderate success - some effect
		random_list = {
			70 = {
				trigger_event = mt_manipulation.0031
				add_gold = 200
			}
			30 = {
				# Caught spreading lies
				trigger_event = mt_manipulation.0028
			}
		}
	}
	else = {
		# Low intrigue - likely to be caught
		random_list = {
			40 = {
				trigger_event = mt_manipulation.0031
				add_gold = 150
			}
			60 = {
				# Exposed as liar
				trigger_event = mt_manipulation.0028
			}
		}
	}

	# Risk: Competitor counter-campaign
	random = {
		chance = 25
		trigger_event = {
			id = mt_manipulation.0029
			days = { 30 90 }
		}
	}
}

####################################
# CLEANUP AND UTILITY EFFECTS
####################################

# Effect: End market manipulation modifier
# Scope: Character
# Usage: <character> = { end_market_manipulation_effect = yes }
end_market_manipulation_effect = {
	if = {
		limit = { has_character_modifier = mt_market_monopoly_active }
		remove_character_modifier = mt_market_monopoly_active
	}
	if = {
		limit = { has_character_modifier = mt_hoarding_goods }
		remove_character_modifier = mt_hoarding_goods
	}
	if = {
		limit = { has_character_modifier = mt_dumping_campaign_active }
		remove_character_modifier = mt_dumping_campaign_active
	}
	if = {
		limit = { has_character_modifier = mt_short_position_active }
		remove_character_modifier = mt_short_position_active
	}
}

# Effect: Regulatory punishment
# Scope: Character
# Usage: <character> = { market_manipulation_punishment_effect = yes }
market_manipulation_punishment_effect = {
	# Heavy fine
	remove_long_term_gold = 500

	# Reputation destroyed
	add_character_modifier = {
		modifier = mt_convicted_manipulator
		years = 20
	}

	# Prestige hit
	add_prestige = -200

	# Possible imprisonment
	random = {
		chance = 30
		imprison = {
			target = root
			type = dungeon
		}
	}

	# End all active manipulations
	end_market_manipulation_effect = yes
}
