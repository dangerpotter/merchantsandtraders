# TRADE HUB STATUS
#######################################
# When a county has 3+ active trade routes, it becomes a trade hub.
# Trade hubs are major economic centers with increased development and prosperity.

	monthly_county_control_growth_add = 0.2
	development_growth_factor = 0.1
	levy_size = 0.1

	# Increased income from becoming a major trade center
	county_modifier = {
		tax_mult = 2.0
	}

	# Trade hubs attract more merchants and artisans
	monthly_prestige = 0.5
}

#######################################
# RULER TRADE POLICIES
#######################################
# Rulers can choose how to interact with merchants passing through their lands.

# HEAVY TAXATION
# Ruler extracts maximum profit from merchants, but damages route health.
# Merchants avoid heavily taxed regions when possible.
trade_taxation_heavy = {
	monthly_county_control_growth_add = -0.1

	# Immediate profit but reduced long-term trade
	county_modifier = {
		tax_mult = 1.0
	}

	# Merchants resent heavy taxation
	monthly_prestige = -0.25
}

# MODERATE TAXATION (BALANCED)
# Standard taxation that doesn't significantly impact merchants.
trade_taxation_moderate = {
	county_modifier = {
		tax_mult = 0.5
	}
}

# TRADE PROTECTION
# Ruler invests in protecting merchants from bandits and rivals.
# Increases route health and makes the county attractive for trade.
trade_protection_active = {
	monthly_county_control_growth_add = 0.2
	development_growth_factor = 0.05

	# Protection costs money but attracts more trade
	county_modifier = {
		tax_mult = -0.5
	}

	# Merchants appreciate protection
	monthly_prestige = 0.5
}
