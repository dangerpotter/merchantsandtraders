		# }

		is_ruler = yes
	}

	effect = {
		# Apply taxation policy modifier
		add_character_modifier = {
			modifier = mt_trade_taxation_policy
			desc = mt_trade_taxation_policy_desc
		}

		# Merchants resent heavy taxation
		add_character_modifier = {
			modifier = mt_merchant_resentment
			years = 10
			desc = mt_merchant_resentment_desc
		}

		# Apply route health penalties to all routes in realm
		# TODO: Implement when route system is ready
		# every_sub_realm_county = {
		#     if = {
		#         limit = { has_county_modifier = active_trade_route_1 }

		#             modifier = route_health_penalty
		#             years = -1  # Permanent until policy revoked
		#         }
		#     }
		# }

		# Notification
		custom_tooltip = establish_trade_taxation_decision_effect_tooltip
	}

	ai_check_interval = 60

		# AI considers this if they want more income
		is_at_war = no
		short_term_gold < 100

		base = 0

		modifier = {
			add = 40
			# Greedy characters more likely
			has_trait = greedy
		}

		modifier = {
			add = 30
			# Struggling financially
			short_term_gold < 50
		}

		modifier = {
			add = 20
			# At war and need funds
			is_at_war = yes
		}

		modifier = {
			add = -50
			# Just rulers less likely
			has_trait = just
		}

		modifier = {
			add = -30
			# High stewardship rulers understand long-term costs

grant_trade_monopoly_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_monopoly.dds"

	desc = grant_trade_monopoly_decision_desc
	selection_tooltip = grant_trade_monopoly_decision_tooltip

	major = yes

	is_shown = {
		# Must be King or Emperor
		highest_held_title_tier >= tier_kingdom

		# Must not already have granted monopoly
		NOT = { has_character_modifier = mt_trade_monopoly_granted }

		# Must not have banned merchants
		NOT = { has_character_modifier = mt_merchant_activity_banned }

		# TODO: Must have at least one merchant in court when merchant system implemented
	}

	is_valid = {
		# Must have prestige to spend
		prestige >= 100

		# TODO: Must select a specific merchant character
		# For now just check they have courtiers
		any_courtier = {
			count >= 1
		}
	}

	is_valid_showing_failures_only = {
		prestige >= 100
	}

	cost = {
		prestige = 100
	}

	effect = {
		# Apply monopoly granted modifier to ruler
		add_character_modifier = {
			modifier = mt_trade_monopoly_granted
			desc = mt_trade_monopoly_granted_desc
		}

		# TODO: When merchant system exists, choose specific merchant and give them monopoly
		# For now, give a random courtier the monopoly modifier
		random_courtier = {
			limit = {
				is_adult = yes
				is_landed = no
			}

			save_scope_as = monopoly_merchant

			add_character_modifier = {
				modifier = mt_trade_monopoly_holder
				desc = mt_trade_monopoly_holder_desc
			}

			# Chosen merchant loves you
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 50
			}
		}

		# All other merchants hate you
		# TODO: Implement when merchant system exists

		#     add_opinion = {
		#         target = root
		#         modifier = angry_opinion
		#         opinion = -30
		#     }
		# }

		# Notification
		custom_tooltip = grant_trade_monopoly_decision_effect_tooltip
	}

	ai_check_interval = 120

	ai_potential = {
		# AI rarely does this
		is_at_war = no
	}

		base = 0

			add = 20
			# Very high opinion of a specific merchant
			any_courtier = {
				opinion = {
					target = root
					value >= 80
				}
			}
		}

			add = -80

revoke_trade_monopoly_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_monopoly.dds"

	desc = revoke_trade_monopoly_decision_desc
	selection_tooltip = revoke_trade_monopoly_decision_tooltip

	is_shown = {
		# Must have granted monopoly
		has_character_modifier = mt_trade_monopoly_granted
	}

	is_valid = {
		is_ruler = yes
	}

	effect = {
		# Remove monopoly from ruler
		remove_character_modifier = mt_trade_monopoly_granted

		# Find and remove monopoly from holder
		# TODO: Need to track who has the monopoly properly
		every_courtier = {
			limit = {
				has_character_modifier = mt_trade_monopoly_holder
			}

			remove_character_modifier = mt_trade_monopoly_holder

			# They're angry about losing monopoly
			add_opinion = {
				target = root
				modifier = angry_opinion
				opinion = -30
			}
		}

		# Other merchants are pleased
		add_character_modifier = {
			modifier = mt_merchant_approval
			years = 5
			desc = mt_merchant_approval_desc
		}

		custom_tooltip = revoke_trade_monopoly_decision_effect_tooltip
	}

	ai_check_interval = 60

	ai_will_do = {
		base = 0

		modifier = {
			add = 40
			# If the monopoly holder has low opinion
			any_courtier = {
				has_character_modifier = mt_trade_monopoly_holder
				opinion = {
					target = root
					value < 20
				}
			}
		}
	}
}

# DECISION 5: BAN MERCHANT ACTIVITY
####################################

ban_merchant_activity_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_realm.dds"

	desc = ban_merchant_activity_decision_desc
	selection_tooltip = ban_merchant_activity_decision_tooltip

	major = yes

	is_shown = {
		# Must be a landed ruler
		is_landed = yes

		# Must not already have banned merchants
		NOT = { has_character_modifier = mt_merchant_activity_banned }
	}

	is_valid = {
		is_ruler = yes

		# This is a major decision with consequences
		custom_tooltip = {
			text = ban_merchant_activity_decision_warning
			always = yes
		}
	}

	effect = {
		# Remove any existing trade policies first
		if = {
			limit = { has_character_modifier = mt_trade_taxation_policy }
			remove_character_modifier = mt_trade_taxation_policy
		}

		if = {
			limit = { has_character_modifier = mt_merchant_protection_policy }
			remove_character_modifier = mt_merchant_protection_policy
		}

		if = {
			limit = { has_character_modifier = mt_trade_monopoly_granted }
			remove_character_modifier = mt_trade_monopoly_granted
		}

		if = {
			limit = { has_character_modifier = mt_foreign_merchants_active }
			remove_character_modifier = mt_foreign_merchants_active
		}

		# Apply merchant ban modifier
		add_character_modifier = {
			modifier = mt_merchant_activity_banned
			desc = mt_merchant_activity_banned_desc
		}

		# All merchants furious
		add_character_modifier = {
			modifier = mt_merchant_fury
			years = 20
			desc = mt_merchant_fury_desc
		}

		# TODO: Close all trade posts and routes when system exists
		# every_sub_realm_county = {
		#     remove_all_trade_routes_effect = yes
		# }

		# Notification
		custom_tooltip = ban_merchant_activity_decision_effect_tooltip
	}

	ai_check_interval = 120

	ai_potential = {
		# AI almost never does this
		is_at_war = no
	}

	ai_will_do = {
		base = 0

		modifier = {
			add = 5
			# Isolationist cultures might do this
			# TODO: Check for isolationist culture trait when defined
			culture = {
				# has_cultural_parameter = isolationist
			}
		}

		modifier = {
			add = -95
			# Base chance is extremely low
			always = yes
		}
	}
}
