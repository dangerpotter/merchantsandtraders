##############################################
# HIRE EMPLOYEE DECISION
# Allows merchants to recruit staff for their company
##############################################
hire_employee_decision = {
	title = hire_employee_decision_title
	desc = hire_employee_decision_desc
	selection_tooltip = hire_employee_decision_tooltip

	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"

	sort_order = 100

	is_shown = {
		has_trait = mt_merchant
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable = no
		is_at_war = no

		# Must have enough gold for hiring fee and salary
		custom_tooltip = {
			text = hire_employee_gold_requirement_tt
			gold >= 60
		}

		# Cannot exceed maximum employees (10 total)
		custom_tooltip = {
			text = hire_employee_max_employees_tt
			OR = {
				NOT = { has_variable = total_employees }
				var:total_employees < 10
			}
		}
	}

	cost = {
		gold = 50
	}

	effect = {
		# Trigger hiring event to select role and quality
		trigger_event = {
			id = mt_company_management_events.1
		}

		# Increment total employees counter
		if = {
			limit = { has_variable = total_employees }
			change_variable = {
				name = total_employees
				add = 1
			}
		}
		else = {
			set_variable = {
				name = total_employees
				value = 1
			}
		}
	}

	ai_check_interval = 60

	ai_potential = {
		has_trait = mt_merchant
		gold >= 100
	}

	ai_will_do = {
		base = 20

		modifier = {
			add = 20
			gold >= 200
		}

		modifier = {
			add = 10
			has_trait = ambitious
		}

		modifier = {
			add = -50
			has_variable = total_employees
			var:total_employees >= 5
		}

		modifier = {
			factor = 0
			has_variable = total_employees
			var:total_employees >= 8
		}
	}
}

fire_employee_decision = {
	title = fire_employee_decision_title
	desc = fire_employee_decision_desc
	selection_tooltip = fire_employee_decision_tooltip

	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"

	is_shown = {
		has_trait = mt_merchant

		# Must have at least one employee
		any_courtier = {
			has_variable = employer
			var:employer = root
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable = no
	}

	effect = {
		# Trigger firing selection event
		trigger_event = {
			id = mt_company_management_events.41
		}

		# Decrement total employees counter
		if = {
			limit = {
				has_variable = total_employees
				var:total_employees > 0
			}
			change_variable = {
				name = total_employees
				add = -1
			}
		}
	}

	ai_check_interval = 60

	ai_potential = {
		has_trait = mt_merchant

		# AI only fires employees with very low loyalty or performance
		any_courtier = {
			has_variable = employer
			var:employer = root
			OR = {
				AND = {
					has_variable = employee_loyalty
					var:employee_loyalty < 20
				}
				AND = {
					has_variable = employee_performance
					var:employee_performance < 30
				}
			}
		}
	}

	ai_will_do = {
		base = 30

		modifier = {
			add = 30
			has_trait = callous
		}

		modifier = {
			add = 20
			has_trait = wrathful
		}

		modifier = {
			add = -30
			has_trait = compassionate
		}

		modifier = {
			add = -20
			has_trait = forgiving
		}
	}
}

##############################################
# INVESTIGATE EMBEZZLEMENT DECISION
# Check for employee fraud
##############################################
investigate_embezzlement_decision = {
	title = investigate_embezzlement_decision_title
	desc = investigate_embezzlement_decision_desc
	selection_tooltip = investigate_embezzlement_decision_tooltip

	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"

	sort_order = 104

	cooldown = { years = 1 }

	is_shown = {
		has_trait = mt_merchant

		# Must have employees
		any_courtier = {
			has_variable = employer
			var:employer = root
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable = no

		# Must have gold for investigation
		custom_tooltip = {
			text = investigate_embezzlement_gold_requirement_tt
			gold >= 30
		}
	}

	cost = {
		gold = 30
	}

	effect = {
		# Check each employee for embezzlement
		every_courtier = {
			limit = {
				has_variable = employer
				var:employer = root
				has_variable = employee_loyalty
				var:employee_loyalty < 40
			}

			save_temporary_scope_as = suspect_employee

			root = {
				embezzlement_detection_effect = {
					employee = scope:suspect_employee
				}
			}
		}

		# All loyal employees lose a bit of trust
		every_courtier = {
			limit = {
				has_variable = employer
				var:employer = root
				has_variable = employee_loyalty
				var:employee_loyalty >= 60
			}

			change_variable = {
				name = employee_loyalty
				add = -5
			}
		}

		# Trigger investigation results event
		trigger_event = {
			id = mt_company_management_events.43
			days = 7
		}
	}

	ai_check_interval = 120

	ai_potential = {
		has_trait = mt_merchant

		# AI investigates if they suspect embezzlement
		any_courtier = {
			has_variable = employer
			var:employer = root
			has_variable = employee_loyalty
			var:employee_loyalty < 25
		}
	}

	ai_will_do = {
		base = 40

		modifier = {
			add = 30
			has_trait = paranoid
		}

		modifier = {
			add = 20
			has_trait = diligent
		}

		modifier = {
			add = 20
			has_trait = just
		}

		modifier = {
			add = -20
			has_trait = trusting
		}

		modifier = {
			add = -30
			has_trait = forgiving
		}
	}
}

	is_shown = {
		has_trait = mt_merchant

		}
	}

	cost = {
		gold = 100
		prestige = 150
	}

	effect = {
		# Improve all employees
		every_courtier = {
			limit = {
				has_variable = employer
				var:employer = root
			}

			# Increase loyalty
			change_variable = {
				name = employee_loyalty
				add = 10
			}

			# Increase performance
			change_variable = {
				name = employee_performance
				add = 10
			}

			# Small chance to improve quality
			random = {
				chance = 20

				if = {
					limit = { var:employee_quality < 4 }
					change_variable = {
						name = employee_quality
						add = 1
					}
				}
			}
		}

		# Trigger training event
		trigger_event = {
			id = mt_company_management_events.45
			days = 30
		}
	}

	ai_check_interval = 180

		gold >= 300
	}

	ai_will_do = {
		base = 15

		modifier = {
			add = 25
			has_trait = diligent
		}

		modifier = {
			add = 20
			has_trait = ambitious
		}

		modifier = {
			add = 15
			learning >= 15
		}

		modifier = {
			add = -20
			has_trait = greedy
		}

		modifier = {
			factor = 0
			gold < 250
		}
