# Trade Espionage System - Decisions
# Player decisions for managing spies and espionage operations

# Recruit a spy to work for you
recruit_spy_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"

	desc = recruit_spy_decision_desc
	selection_tooltip = recruit_spy_decision_tooltip

	is_shown = {
		# Must be adult and have at least some gold
		is_adult = yes
		gold >= 50
	}

	is_valid = {
		gold >= 50
		# Can have multiple spies, but limit to 5
		OR = {
			NOT = { has_variable = spy_count }
			var:spy_count < 5
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
	}

	cost = {
		gold = 50
	}

	effect = {
		custom_tooltip = recruit_spy_decision_effect_tooltip

		# Recruit the spy
		recruit_spy_effect = yes

		# Trigger recruitment event
		trigger_event = {
			id = mt_espionage.1
			days = 3
		}

		# Add modifier showing you have spies
		if = {
			limit = {
				NOT = { has_character_modifier = mt_employs_spies }
			}
			add_character_modifier = {
				modifier = mt_employs_spies
				years = 10
			}
		}
	}

	ai_check_interval = 60

	ai_potential = {
		is_adult = yes
		gold >= 100
		intrigue >= 10
	}

	ai_will_do = {
		base = 10

		modifier = {
			factor = 2
			intrigue >= 15
		}

		modifier = {
			factor = 2
			has_trait = schemer
		}

		modifier = {
			factor = 0.5
			has_trait = honest
		}

		modifier = {
			factor = 0
			gold < 100
		}
	}
}

# Dismiss spy (fire them)
dismiss_spy_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"

	desc = dismiss_spy_decision_desc
	selection_tooltip = dismiss_spy_decision_tooltip

	is_shown = {
		has_variable = has_active_spy
	}

	is_valid = {
		has_variable = has_active_spy
		has_variable = spy_count
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
	}

	effect = {
		custom_tooltip = dismiss_spy_decision_effect_tooltip

		# Pay severance
		remove_short_term_gold = 20

		# Reduce spy count
		if = {
			limit = {
				var:spy_count > 1
			}
			change_variable = {
				name = spy_count
				subtract = 1
			}
		}
		else = {
			remove_variable = spy_count
			remove_variable = has_active_spy
			remove_variable = spy_quality
			remove_variable = spy_loyalty
			remove_variable = spy_success_chance
			remove_variable = spy_salary

			remove_character_modifier = mt_employs_spies
		}

		# Small chance dismissed spy becomes hostile
		random = {
			chance = 15
			modifier = {
				factor = 2
				var:spy_loyalty < 50
			}

			trigger_event = mt_espionage.111
		}

		send_interface_toast = {
			title = dismiss_spy_decision_toast
			left_icon = root

			custom_tooltip = dismiss_spy_decision_toast_desc
		}
	}

	ai_check_interval = 90

	ai_potential = {
		has_variable = has_active_spy
	}

	ai_will_do = {
		base = 5

		modifier = {
			factor = 5
			gold < 30
		}

		modifier = {
			factor = 3
			has_variable = spy_loyalty
			var:spy_loyalty < 30
		}

		modifier = {
			factor = 0
			gold >= 100
		}
	}
}
