##############################################
# CORNER THE MARKET
# Attempt to monopolize a specific trade good
##############################################
attempt_market_corner_decision = {
	title = attempt_market_corner_decision_title
	desc = attempt_market_corner_decision_desc
	selection_tooltip = attempt_market_corner_decision_tooltip

	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"

	major = yes

	is_shown = {
		is_any_merchant_trigger = yes

		# Must have significant merchant reputation
		OR = {
			has_character_modifier = mt_reputation_prominent_merchant
			has_character_modifier = mt_reputation_renowned_trader
			has_character_modifier = mt_reputation_merchant_magnate
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable = no
		is_at_war = no

		# Must not already have an active monopoly
		custom_tooltip = {
			text = attempt_market_corner_no_active_monopoly_tt
			NOT = { has_character_modifier = mt_market_monopoly_active }
		}

		# Must have substantial wealth
		custom_tooltip = {
			text = attempt_market_corner_wealth_requirement_tt
			gold >= 1000
		}

		# Must have high stewardship or greed
		custom_tooltip = {
			text = attempt_market_corner_skill_requirement_tt
			OR = {
				stewardship >= 12
				has_trait = greedy
				has_trait = ambitious
			}
		}
	}

	cost = {
		gold = 1000
	}

	effect = {
		# Apply the corner market effect
		corner_market_effect = yes

		# Add stress if compassionate or just
		if = {
			limit = { has_trait = compassionate }
			add_stress = 100
		}
		if = {
			limit = { has_trait = just }
			add_stress = 75
		}

		# Reduce stress if greedy or ambitious
		if = {
			limit = { has_trait = greedy }
			add_stress = -50
		}
		if = {
			limit = { has_trait = ambitious }
			add_stress = -25
		}

		# Trigger immediate event
		trigger_event = mt_manipulation.0001
	}

	ai_potential = {
		is_at_war = no
		gold >= 2000
	}

	ai_will_do = {
		base = 0

		modifier = {
			add = 30
			has_trait = greedy
		}
		modifier = {
			add = 20
			has_trait = ambitious
		}
		modifier = {
			add = -100
			has_trait = just
		}
		modifier = {
			add = -100
			has_trait = compassionate
		}
	}
}

		}

	}

			add = 40
			has_trait = greedy
		}

			add = -100
			has_trait = compassionate
		}
		modifier = {
			add = -50
			has_trait = just
		}
	}

# MARKET FLOODING ATTACK
# Dump goods to crash prices and bankrupt rivals
##############################################
market_flooding_decision = {
	title = market_flooding_decision_title
	desc = market_flooding_decision_desc
	selection_tooltip = market_flooding_decision_tooltip

	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"

	is_shown = {
		is_any_merchant_trigger = yes
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable = no

		# Must not already be dumping
		custom_tooltip = {
			text = market_flooding_not_active_tt
			NOT = { has_character_modifier = mt_dumping_campaign_active }
		}

		# Must have capital to absorb losses
		custom_tooltip = {
			text = market_flooding_wealth_requirement_tt
			gold >= 400
		}

		# Should have a rival or competitor
		custom_tooltip = {
			text = market_flooding_has_rival_tt
			any_relation = {
				type = rival
				is_any_merchant_trigger = yes
			}
		}
	}

	cost = {
		gold = 300
	}

	effect = {
		# Apply the dumping effect
		dump_goods_effect = yes

		# Stress impacts
		if = {
			limit = { has_trait = just }
			add_stress = 100
		}
		if = {
			limit = { has_trait = vengeful }
			add_stress = -50
		}

		# Trigger flooding event
		trigger_event = mt_manipulation.0017
	}

	ai_potential = {
		gold >= 800
		any_relation = {
			type = rival
			is_any_merchant_trigger = yes
		}
	}

	ai_will_do = {
		base = 0

		modifier = {
			add = 50
			has_trait = vengeful
		}
		modifier = {
			add = 25
			has_trait = ambitious
		}
		modifier = {
			add = -100
			has_trait = just
		}
	}
}

	desc = form_price_cartel_decision_desc
	selection_tooltip = form_price_cartel_decision_tooltip

	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"

	major = yes

	is_shown = {
		is_any_merchant_trigger = yes

		# Must have good reputation
		OR = {
			has_character_modifier = mt_reputation_respected_trader
			has_character_modifier = mt_reputation_prominent_merchant
			has_character_modifier = mt_reputation_renowned_trader
			has_character_modifier = mt_reputation_merchant_magnate
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable = no

		# Must not already be in a cartel
		custom_tooltip = {
			text = form_cartel_not_member_tt
			NOT = { has_character_modifier = mt_cartel_member }
		}

		# Must have other merchants to collude with
		custom_tooltip = {
			text = form_cartel_has_merchants_tt
			any_living_character = {
				count >= 2
				is_any_merchant_trigger = yes
				NOT = { this = root }
				opinion = {
					target = root
					value >= 25
				}
			}
		}
	}

	cost = {
		gold = 200
		prestige = 100
	}

	effect = {
		# Apply cartel effect
		collusion_effect = yes

		# Stress impacts
		if = {
			limit = { has_trait = just }
			add_stress = 125
		}
		if = {
			limit = { has_trait = honest }
			add_stress = 100
		}
		if = {
			limit = { has_trait = greedy }
			add_stress = -50
		}

		# Trigger cartel formation event
		trigger_event = mt_manipulation.0033
	}

	ai_potential = {
		gold >= 500
	}

	ai_will_do = {
		base = 0

		modifier = {
			add = 40
			has_trait = greedy
		}
		modifier = {
			add = 30
			has_trait = ambitious
		}
		modifier = {
			add = -100
			has_trait = just
		}

			add = -100
