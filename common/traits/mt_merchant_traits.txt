	trait_tier = 1

	diplomacy = 2

	monthly_character_income_mult = 0.05

	same_opinion = 5

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { always = yes }
				desc = wandering_merchant_desc

	ai_boldness = 5
